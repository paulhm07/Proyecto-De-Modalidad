    82→    82→function Cajita({
    83→    83→  producto,
    84→    84→  cantidad,
    85→    85→  cargada,
    86→    86→  draggable,
    87→    87→  onClick,
    88→    88→  onDragStart,
    89→    89→  size = "md",
    90→    90→  index,
    91→    91→}: {
    92→    92→  producto: ProductoVisual;
    93→    93→  cantidad: number;
    94→    94→  cargada?: boolean;
    95→    95→  draggable?: boolean;
    96→    96→  onClick?: () => void;
    97→    97→  onDragStart?: () => void;
    98→    98→  size?: CajitaSize;
    99→    99→  index: number;
   100→   100→}) {
   101→   101→  const emoji = PRODUCTO_EMOJI[producto];
   102→   102→  const cols = cantidad <= 3 ? cantidad : cantidad <= 6 ? 3 : 4;
   103→   103→  const sz = SIZE_MAP[size];
   104→   104→
   105→   105→  return (
   106→   106→    <div
   107→   107→      role={draggable ? "button" : undefined}
   108→   108→      tabIndex={draggable ? 0 : undefined}
   109→   109→      onClick={draggable ? onClick : undefined}
   110→   110→      onKeyDown={(e) => {
   111→   111→        if (!draggable) return;
   112→   112→        if (e.key === "Enter" || e.key === " ") {
   113→   113→          e.preventDefault();
   114→   114→          onClick?.();
   115→   115→        }
   116→   116→      }}
   117→   117→      onDragStart={(e) => {
   118→   118→        if (!draggable) {
   119→   119→          e.preventDefault();
   120→   120→          return;
   121→   121→        }
   122→   122→        e.dataTransfer.setData("text/plain", String(index));
   123→   123→        e.dataTransfer.effectAllowed = "move";
   124→   124→        onDragStart?.();
   125→   125→      }}
   126→   126→      {...(draggable ? ({ draggable: true } as Record<string, unknown>) : {})}
   127→   127→      className={`relative grid shrink-0 select-none place-items-center rounded-lg border-2 shadow-sm transition ${sz.box} ${
   128→   128→        cargada
   129→   129→          ? "border-emerald-500 bg-emerald-50"
   130→   130→          : "border-amber-700 bg-amber-100 hover:border-orange-500 hover:bg-amber-200"
   131→   131→      } ${draggable ? "cursor-grab active:cursor-grabbing hover:scale-105" : ""}`}
   132→   132→      style={{ gridTemplateColumns: `repeat(${cols}, 1fr)` }}
   133→   133→      aria-label={`Cajita ${index + 1} con ${cantidad} ${PRODUCTO_NOMBRE[producto]}`}
   134→   134→      title={`Cajita con ${cantidad} ${PRODUCTO_NOMBRE[producto]}`}
   135→   135→    >
   136→   136→      {Array.from({ length: cantidad }).map((_, i) => (
   137→   137→        <span key={i} className={`${sz.emoji} leading-none flex items-center justify-center`}>
   138→   138→          {emoji}
   139→   139→        </span>
   140→   140→      ))}
   141→   141→    </div>
   142→   142→  );
   143→   143→}

   282→   282→
   283→   283→export function CamionMultiplicaciones() {
   284→   284→  const [nivelIdx, setNivelIdx] = useState(0);
   285→   285→  const [estado, setEstado] = useState<Estado>("presentacion");
   286→   286→  const [cajitasCargadas, setCajitasCargadas] = useState<number[]>([]);
   287→   287→  const [camionesDespachados, setCamionesDespachados] = useState(0);
   288→   288→  const [feedback, setFeedback] = useState<string | null>(null);
   289→   289→  const [dragOverCamion, setDragOverCamion] = useState(false);
   290→   290→
   291→   291→  // Ref para timeouts internos (limpieza segura al desmontar)
   292→   292→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   293→   293→  const programar = useCallback((fn: () => void, ms: number) => {
   294→   294→    const t = setTimeout(() => {
   295→   295→      timeoutsRef.current.delete(t);
   296→   296→      fn();
   297→   297→    }, ms);
   298→   298→    timeoutsRef.current.add(t);
   299→   299→  }, []);
   300→   300→
   301→   301→  useEffect(() => {
   302→   302→    const set = timeoutsRef.current;
   303→   303→    return () => {
   304→   304→      set.forEach((t) => clearTimeout(t));
   305→   305→      set.clear();
   306→   306→    };
   307→   307→  }, []);
   308→   308→
   309→   309→  const nivel = nivelesCamion[nivelIdx];
   310→   310→  const avatar = AVATARES_CLIENTES[nivelIdx % AVATARES_CLIENTES.length];
   311→   311→  const totalPila = nivel.grupos + EXTRAS_PILA;
   312→   312→  const indicesPila = useMemo(
   313→   313→    () => Array.from({ length: totalPila }, (_, i) => i),
   314→   314→    [totalPila],
   315→   315→  );
   316→   316→  const cajitasEnPila = indicesPila.filter((i) => !cajitasCargadas.includes(i));
   317→   317→
   318→   318→  /* ---- Transición: celebrando → siguiente nivel o completado ---- */
   319→   319→  useEffect(() => {
   320→   320→    if (estado !== "celebrando") return;
   321→   321→    const t = setTimeout(() => {
   322→   322→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   323→   323→        setEstado("completado");
   324→   324→      } else {
   325→   325→        setNivelIdx((n) => n + 1);
   326→   326→        setCajitasCargadas([]);
   327→   327→        setFeedback(null);
   328→   328→        setEstado("presentacion");
   329→   329→      }
   330→   330→    }, 2800);
   331→   331→    return () => clearTimeout(t);
   332→   332→  }, [estado, nivelIdx]);
   333→   333→
   334→   334→  /* ---- Acciones ---- */
   335→   335→
   336→   336→  const empezarCarga = useCallback(() => {
   337→   337→    setEstado("cargando");
   338→   338→    setFeedback(null);
   339→   339→  }, []);
   340→   340→
   341→   341→  const cargarCajita = useCallback(
   342→   342→    (i: number) => {
   343→   343→      if (cajitasCargadas.includes(i)) return;
   344→   344→      setCajitasCargadas((p) => [...p, i]);
   345→   345→      setFeedback(null);
   346→   346→    },
   347→   347→    [cajitasCargadas],
   348→   348→  );
   349→   349→
   350→   350→  const quitarCajita = useCallback((i: number) => {
   351→   351→    setCajitasCargadas((p) => p.filter((x) => x !== i));
   352→   352→    setFeedback(null);
   353→   353→  }, []);
   354→   354→
   355→   355→  const vaciarCamion = useCallback(() => {
   356→   356→    setCajitasCargadas([]);
   357→   357→    setFeedback(null);
   358→   358→  }, []);
   359→   359→
   360→   360→  const despachar = useCallback(() => {
   361→   361→    if (estado !== "cargando") return;
   362→   362→    setEstado("verificando");
   363→   363→    const correcto = cajitasCargadas.length === nivel.grupos;
   364→   364→    programar(() => {
   365→   365→      if (correcto) {
   366→   366→        setCamionesDespachados((v) => v + 1);
   367→   367→        setEstado("celebrando");
   368→   368→      } else {
   369→   369→        setFeedback(nivel.feedback_error);
   370→   370→        setEstado("cargando");
   371→   371→      }
   372→   372→    }, 450);
   373→   373→  }, [estado, cajitasCargadas.length, nivel.grupos, nivel.feedback_error, programar]);
   374→   374→
   375→   375→  const reiniciarJuego = useCallback(() => {
   376→   376→    setNivelIdx(0);
   377→   377→    setEstado("presentacion");
   378→   378→    setCajitasCargadas([]);
   379→   379→    setCamionesDespachados(0);
   380→   380→    setFeedback(null);
   381→   381→  }, []);
   382→   382→
   383→   383→  /* ---- Drag handlers para el camión ---- */
   384→   384→  const onDropCamion = useCallback(
   385→   385→    (e: React.DragEvent) => {
   386→   386→      e.preventDefault();
   387→   387→      setDragOverCamion(false);
   388→   388→      const idx = Number(e.dataTransfer.getData("text/plain"));
   389→   389→      if (!Number.isNaN(idx)) cargarCajita(idx);
   390→   390→    },
   391→   391→    [cargarCajita],
   392→   392→  );
   393→   393→
   394→   394→  /* ---- Derivados ---- */
   395→   395→  const progresoNivel = useMemo(
   396→   396→    () => ((nivelIdx + (estado === "celebrando" ? 1 : 0)) / TOTAL_NIVELES) * 100,
   397→   397→    [nivelIdx, estado],
   398→   398→  );
   399→   399→  const esPresentacion = estado === "presentacion";
   400→   400→  const esCargando = estado === "cargando";
   401→   401→  const esVerificando = estado === "verificando";
   402→   402→  const esCelebrando = estado === "celebrando";
   403→   403→  const puedeDespachar = esCargando && cajitasCargadas.length > 0;
   404→   404→
   405→   405→  /* ---- Tamaños de cajitas según cantidad de grupos ---- */
   406→   406→  const sizePila: CajitaSize =
   407→   407→    nivel.grupos > 12 ? "sm" : nivel.grupos > 6 ? "md" : "lg";
   408→   408→  const sizeCamion: CajitaSize = nivel.grupos > 6 ? "sm" : "md";
   409→   409→
   410→   410→  /* ===================== RENDER ===================== */
   411→   411→
   412→   412→  return (
   413→   413→    <div className="relative min-h-[calc(100vh-72px)] overflow-hidden bg-gradient-to-br from-amber-100 via-orange-200 to-rose-200 font-display">
   414→   414→      {/* Toldos de mercado arriba */}
   415→   415→      <ToldosMercado />
   416→   416→
   417→   417→      {/* Decoración de fondo: mercado nicaragüense */}
   418→   418→      <div aria-hidden className="pointer-events-none absolute inset-0 opacity-[0.07]">
   419→   419→        <div className="absolute left-4 top-24 text-7xl">🥭</div>
   420→   420→        <div className="absolute right-8 top-32 text-7xl">🌽</div>
   421→   421→        <div className="absolute left-10 bottom-24 text-7xl">🪅</div>
   422→   422→        <div className="absolute right-12 bottom-32 text-7xl">🥖</div>
   423→   423→        <div className="absolute left-1/3 top-12 text-6xl">☕</div>
   424→   424→        <div className="absolute right-1/3 bottom-16 text-6xl">🍫</div>
   425→   425→      </div>
   426→   426→
   427→   427→      <div className="relative z-10 mx-auto max-w-5xl px-3 pb-6 pt-14 sm:px-5 sm:pt-16">
   428→   428→        {/* ===================== HUD superior ===================== */}
   429→   429→        <div className="mb-4 rounded-2xl border-2 border-teal-200 bg-white/70 p-3 shadow-md backdrop-blur sm:p-4">
