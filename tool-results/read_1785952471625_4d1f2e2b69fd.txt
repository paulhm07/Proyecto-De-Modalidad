    65→/* ----------------------- Helpers ----------------------- */
    66→
    67→function limpiar(token: string): string {
    68→  return token
    69→    .replace(/^[.,;:!¡¿()"']+/, "")
    70→    .replace(/[.,;:!¡¿()"']+$/, "");
    71→}
    72→
    73→function normalizar(s: string): string {
    74→  return s
    75→    .toLowerCase()
    76→    .trim()
    77→    .split(/\s+/)
    78→    .map(limpiar)
    79→    .filter(Boolean)
    80→    .join(" ");
    81→}
    82→
    83→function esNivelPronombres(tipo: NivelBus["tipo_ejercicio"]): boolean {
    84→  return tipo === "pronombre_el_ella" || tipo === "pronombre_ellos_ellas";
    85→}
    86→
    87→function tokenizarEnunciado(enunciado: string): string[] {
    88→  return enunciado
    89→    .split(/\s+/)
    90→    .map((t) => limpiar(t))
    91→    .filter(Boolean);
    92→}
    93→
    94→function mezclar<T>(arr: readonly T[]): T[] {
    95→  const a = [...arr];
    96→  for (let i = a.length - 1; i > 0; i--) {
    97→    const j = Math.floor(Math.random() * (i + 1));
    98→    [a[i], a[j]] = [a[j], a[i]];
    99→  }
   100→  return a;
   101→}

  1643→/* ============================================================
  1644→   Helper: resalta la palabra correcta en el enunciado al celebrar
  1645→   ============================================================ */
  1646→function resaltarPalabraCorrecta(
  1647→  enunciado: string,
  1648→  palabraCorrecta: string,
  1649→  celebrando: boolean,
  1650→): React.ReactNode {
  1651→  if (!celebrando) return enunciado;
  1652→
  1653→  // Normalizar para búsqueda insensible a mayúsculas/puntuación
  1654→  const tokens = enunciado.split(/(\s+)/); // conserva espacios
  1655→  const correctasNorm = normalizar(palabraCorrecta)
  1656→    .split(" ")
  1657→    .filter(Boolean);
  1658→
  1659→  if (correctasNorm.length === 0) return enunciado;
  1660→
  1661→  // Para 1 palabra: marcar el primer token que coincida
  1662→  if (correctasNorm.length === 1) {
  1663→    return tokens.map((tok, i) => {
  1664→      if (normalizar(tok) === correctasNorm[0]) {
  1665→        return (
  1666→          <mark
  1667→            key={i}
  1668→            className="rounded bg-emerald-200 px-1 font-bold text-emerald-900 shadow-[0_0_8px_rgba(16,185,129,0.6)]"
  1669→          >
  1670→            {tok}
  1671→          </mark>
  1672→        );
  1673→      }
  1674→      return tok;
  1675→    });
  1676→  }
  1677→
  1678→  // Para múltiples palabras: marcar todos los tokens que coincidan
  1679→  return tokens.map((tok, i) => {
  1680→    if (correctasNorm.includes(normalizar(tok))) {
  1681→      return (
  1682→        <mark
  1683→          key={i}
  1684→          className="rounded bg-emerald-200 px-1 font-bold text-emerald-900 shadow-[0_0_8px_rgba(16,185,129,0.6)]"
  1685→        >
  1686→          {tok}
  1687→        </mark>
  1688→      );
  1689→    }
  1690→    return tok;
  1691→  });
  1692→}
