   162→export function ContenidoMINEDViewer() {
   163→  const { setVista, mostrarToast } = useApp();
   164→  const [tab, setTab] = useState<TabModulo>("m1");
   165→  const [manifest, setManifest] = useState<Manifest | null>(null);
   166→  const [m1, setM1] = useState<{ problemas: ProblemaMat[]; titulo: string } | null>(null);
   167→  const [m2, setM2] = useState<{ desafios: DesafioVisual[]; titulo: string } | null>(null);
   168→  const [m3, setM3] = useState<{ lecturas: Lectura[]; titulo: string } | null>(null);
   169→  const [m4, setM4] = useState<{
   170→    palabras: PalabraOrt[];
   171→    oraciones: OracionSigno[];
   172→    titulo: string;
   173→  } | null>(null);
   174→  const [cargando, setCargando] = useState(true);
   175→
   176→  useEffect(() => {
   177→    let cancelado = false;
   178→    (async () => {
   179→      setCargando(true);
   180→      try {
   181→        const [manRes, m1Res, m2Res, m3Res, m4Res] = await Promise.all([
   182→          fetch("/contenido-mined/index.json"),
   183→          fetch("/contenido-mined/modulo1-multiplicacion-division.json"),
   184→          fetch("/contenido-mined/modulo2-fracciones-geometria.json"),
   185→          fetch("/contenido-mined/modulo3-comprension-sintaxis.json"),
   186→          fetch("/contenido-mined/modulo4-ortografia-signos.json"),
   187→        ]);
   188→        if (cancelado) return;
   189→        const [man, d1, d2, d3, d4] = await Promise.all([
   190→          manRes.json(),
   191→          m1Res.json(),
   192→          m2Res.json(),
   193→          m3Res.json(),
   194→          m4Res.json(),
   195→        ]);
   196→        setManifest(man);
   197→        setM1({ problemas: d1.problemas, titulo: d1.titulo });
   198→        setM2({ desafios: d2.desafios, titulo: d2.titulo });
   199→        setM3({ lecturas: d3.lecturas, titulo: d3.titulo });
   200→        setM4({
   201→          palabras: d4.palabras,
   202→          oraciones: d4.oraciones,
   203→          titulo: d4.titulo,
   204→        });
   205→      } catch (err) {
   206→        if (!cancelado) {
   207→          mostrarToast(
   208→            err instanceof Error ? err.message : "Error al cargar contenido MINED",
   209→            "error",
   210→          );
   211→        }
   212→      } finally {
   213→        if (!cancelado) setCargando(false);
   214→      }
   215→    })();
   216→    return () => {
   217→      cancelado = true;
   218→    };
   219→  }, [mostrarToast]);
   220→
   221→  const copiarJSON = async (modulo: TabModulo) => {
   222→    const archivo = {
   223→      m1: "modulo1-multiplicacion-division.json",
   224→      m2: "modulo2-fracciones-geometria.json",
   225→      m3: "modulo3-comprension-sintaxis.json",
   226→      m4: "modulo4-ortografia-signos.json",
   227→    }[modulo];
   228→    try {
   229→      const res = await fetch(`/contenido-mined/${archivo}`);
   230→      const text = await res.text();
   231→      await navigator.clipboard.writeText(text);
   232→      mostrarToast(`JSON del módulo copiado al portapapeles`, "exito");
   233→    } catch {
   234→      mostrarToast("No se pudo copiar el JSON", "error");
   235→    }
   236→  };
   237→
   238→  const descargarJSON = (modulo: TabModulo) => {
   239→    const archivo = {
   240→      m1: "modulo1-multiplicacion-division.json",
   241→      m2: "modulo2-fracciones-geometria.json",
   242→      m3: "modulo3-comprension-sintaxis.json",
   243→      m4: "modulo4-ortografia-signos.json",
   244→    }[modulo];
   245→    const a = document.createElement("a");
   246→    a.href = `/contenido-mined/${archivo}`;
   247→    a.download = archivo;
   248→    a.click();
   249→    mostrarToast(`Descargando ${archivo}`, "info");
   250→  };
   251→
   252→  if (cargando || !manifest) {
   253→    return (
   254→      <div className="mx-auto flex max-w-5xl items-center justify-center px-4 py-20">
   255→        <div className="flex flex-col items-center gap-3">
   256→          <div className="h-10 w-10 animate-spin rounded-full border-4 border-cyan-400/30 border-t-cyan-400" />
   257→          <p className="text-sm font-bold text-cyan-200">Cargando contenido MINED…</p>
   258→        </div>
   259→      </div>
   260→    );
   261→  }
   262→
   263→  return (
   264→    <div className="mx-auto max-w-5xl px-4 py-6 sm:py-8">
   265→      {/* Botón volver */}
   266→      <button
   267→        onClick={() => setVista("maestro")}
   268→        className="subject-back-btn mb-4 inline-flex items-center gap-1.5 px-3 py-2 text-sm"
   269→      >
   270→        <ArrowLeft size={16} strokeWidth={2.5} /> Volver al panel
   271→      </button>
   272→
   273→      {/* Hero */}
   274→      <section className="subject-hero mb-6 rounded-3xl p-6">
   275→        <div className="flex items-start gap-4">
   276→          <div className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-400 to-orange-600 shadow-lg ring-2 ring-white/30">
   277→            <BookOpen size={28} className="text-white" strokeWidth={2.5} />
   278→          </div>
   279→          <div className="flex-1 min-w-0">
   280→            <p className="label-on-dark text-xs uppercase tracking-widest">
   281→              {manifest.curriculum} · {manifest.grado} · {manifest.semestre}
   282→            </p>
   283→            <h1 className="subject-hero-title mt-1 text-3xl sm:text-4xl">
   284→              Banco de Contenido MINED
   285→            </h1>
   286→            <p className="subject-hero-subtitle mt-2 text-sm sm:text-base">
   287→              {manifest.descripcion}
   288→            </p>
   289→            <div className="mt-3 flex flex-wrap gap-2">
   290→              {manifest.elementos_culturales_nicaraguenses.slice(0, 6).map((el, i) => (
   291→                <span
   292→                  key={i}
   293→                  className="rounded-full bg-white/10 px-2.5 py-1 text-[11px] font-bold text-cyan-100 ring-1 ring-cyan-300/30"
   294→                >
   295→                  {el}
   296→                </span>
   297→              ))}
   298→            </div>
   299→          </div>
   300→        </div>
   301→      </section>
   302→
   303→      {/* Tabs de módulos */}
   304→      <div className="mb-5 grid grid-cols-2 gap-2 sm:grid-cols-4">
   305→        {(["m1", "m2", "m3", "m4"] as TabModulo[]).map((m) => {
   306→          const Icon = ICONOS_MODULO[m];
   307→          const c = COLORES_MODULO[m];
   308→          const activo = tab === m;
   309→          const mod = manifest.modulos[Number(m.slice(1)) - 1];
   310→          return (
   311→            <button
   312→              key={m}
   313→              onClick={() => setTab(m)}
   314→              className={`flex flex-col items-center gap-1.5 rounded-2xl p-3 text-center transition-all ${
   315→                activo
   316→                  ? `bg-gradient-to-br ${c.grad} text-white shadow-lg ring-2 ring-white/40`
   317→                  : `${c.bg} ${c.text} ring-1 ${c.ring} hover:scale-[1.02]`
   318→              }`}
   319→            >
   320→              <Icon size={22} strokeWidth={2.5} />
   321→              <span className="text-[11px] font-black uppercase tracking-wide">
   322→                Módulo {m.slice(1)}
   323→              </span>
   324→              <span className={`text-[10px] font-semibold ${activo ? "text-white/80" : "opacity-70"}`}>
   325→                {mod.asignatura === "Matemática" ? "Mat" : "Lengua"}
   326→              </span>
   327→            </button>
   328→          );
   329→        })}
   330→      </div>
   331→
   332→      {/* Acciones de exportación */}
   333→      <div className="mb-5 flex flex-wrap gap-2">
   334→        <button
   335→          onClick={() => copiarJSON(tab)}
   336→          className="glass-on-dark inline-flex items-center gap-1.5 rounded-xl px-3 py-2 text-xs font-bold text-cyan-100 ring-1 ring-cyan-300/30 transition-all hover:scale-105"
   337→        >
   338→          <Copy size={13} strokeWidth={2.5} /> Copiar JSON
   339→        </button>
   340→        <button
   341→          onClick={() => descargarJSON(tab)}
   342→          className="glass-on-dark inline-flex items-center gap-1.5 rounded-xl px-3 py-2 text-xs font-bold text-cyan-100 ring-1 ring-cyan-300/30 transition-all hover:scale-105"
   343→        >
   344→          <Download size={13} strokeWidth={2.5} /> Descargar JSON
   345→        </button>
   346→      </div>
   347→
   348→      {/* Contenido según tab */}
   349→      {tab === "m1" && m1 && <Modulo1View problemas={m1.problemas} titulo={m1.titulo} />}
   350→      {tab === "m2" && m2 && <Modulo2View desafios={m2.desafios} titulo={m2.titulo} />}
   351→      {tab === "m3" && m3 && <Modulo3View lecturas={m3.lecturas} titulo={m3.titulo} />}
   352→      {tab === "m4" && m4 && (
   353→        <Modulo4View palabras={m4.palabras} oraciones={m4.oraciones} titulo={m4.titulo} />
   354→      )}
   355→    </div>
   356→  );
   357→}

   359→// ===== Módulo 1: Multiplicación y División ===================================
   360→
   361→function Modulo1View({ problemas, titulo }: { problemas: ProblemaMat[]; titulo: string }) {
   362→  const [filtro, setFiltro] = useState<"todos" | "multiplicacion" | "division">("todos");
   363→  const filtrados = useMemo(
   364→    () => (filtro === "todos" ? problemas : problemas.filter((p) => p.tipo_operacion === filtro)),
   365→    [problemas, filtro],
   366→  );
   367→
   368→  return (
   369→    <section>
   370→      <div className="challenge-board-sm mb-4 rounded-2xl p-4">
   371→        <div className="flex flex-wrap items-center justify-between gap-3">
   372→          <div>
   373→            <h2 className="challenge-text text-xl">{titulo}</h2>
   374→            <p className="label-on-dark mt-1 text-xs">
   375→              {problemas.length} problemas · 5 multiplicación + 10 división · contexto nicaragüense
   376→            </p>
   377→          </div>
   378→          <div className="flex gap-1.5">
   379→            {(["todos", "multiplicacion", "division"] as const).map((f) => (
   380→              <button
   381→                key={f}
   382→                onClick={() => setFiltro(f)}
   383→                className={`rounded-lg px-3 py-1.5 text-xs font-bold transition-all ${
   384→                  filtro === f
   385→                    ? "bg-cyan-400 text-slate-900"
   386→                    : "glass-on-dark text-cyan-100 ring-1 ring-cyan-300/30"
   387→                }`}
   388→              >
   389→                {f === "todos" ? "Todos" : f === "multiplicacion" ? "× Mult" : "÷ Div"}
   390→              </button>
   391→            ))}
   392→          </div>
   393→        </div>
   394→      </div>
   395→
   396→      <div className="space-y-3">
   397→        {filtrados.map((p, i) => (
   398→          <article
   399→            key={p.id}
   400→            className="mission-card animate-pop rounded-2xl p-4"
   401→            style={{ animationDelay: `${i * 40}ms` }}
   402→          >
   403→            <div className="flex flex-wrap items-start justify-between gap-2">
   404→              <div className="flex items-center gap-2">
   405→                <span
   406→                  className={`mission-num ${
   407→                    p.tipo_operacion === "multiplicacion"
   408→                      ? "mission-num-math"
   409→                      : "mission-num-lang"
   410→                  }`}
   411→                >
   412→                  {p.id.split("-")[1]}
   413→                </span>
   414→                <div>
   415→                  <p className="label-on-dark text-[11px] uppercase tracking-wide">
   416→                    {p.subtipo}
   417→                  </p>
   418→                  <p className="text-xs font-bold text-cyan-200">
   419→                    {p.contexto_cultural}
   420→                  </p>
   421→                </div>
   422→              </div>
   423→              <span
   424→                className={`mission-tag ${
   425→                  p.dificultad === "baja"
   426→                    ? "mission-tag-cyan"
   427→                    : p.dificultad === "media"
   428→                      ? "mission-tag-amber"
   429→                      : "mission-tag-fuchsia"
   430→                }`}
   431→              >
   432→                {p.dificultad}
   433→              </span>
   434→            </div>
   435→
   436→            <p className="challenge-text mt-3 text-base leading-snug">{p.enunciado}</p>
   437→
   438→            <div className="mt-3 flex flex-wrap items-center gap-3">
   439→              <div className="flex items-center gap-2 rounded-xl bg-slate-950/60 px-3 py-2 ring-1 ring-cyan-400/30">
   440→                <Hash size={14} className="text-cyan-300" strokeWidth={2.5} />
   441→                <span className="font-mono text-lg font-black text-cyan-300">
   442→                  {p.operacion_vertical_esperada}
   443→                </span>
   444→              </div>
   445→              <div className="flex items-center gap-1.5">
   446→                <span className="label-on-dark text-[11px]">Respuesta:</span>
   447→                <span className="rounded-lg bg-emerald-400/20 px-2.5 py-1 text-base font-black text-emerald-200 ring-1 ring-emerald-400/50">
   448→                  {p.respuesta_correcta}
   449→                </span>
   450→                {p.residuo > 0 && (
   451→                  <span className="rounded-lg bg-amber-400/20 px-2.5 py-1 text-sm font-black text-amber-200 ring-1 ring-amber-400/50">
   452→                    residuo {p.residuo}
   453→                  </span>
   454→                )}
   455→              </div>
   456→            </div>
   457→
   458→            <div className="mt-3 grid gap-2 sm:grid-cols-3">
   459→              {p.distractores.map((d) => (
   460→                <div
   461→                  key={d}
   462→                  className="rounded-lg bg-rose-500/10 px-2.5 py-1.5 ring-1 ring-rose-400/30"
   463→                >
   464→                  <p className="text-sm font-black text-rose-200">✗ {d}</p>
   465→                  <p className="mt-0.5 text-[10px] leading-tight text-rose-100/70">
   466→                    {p.explicacion_distractores[String(d)] ?? "Distractor numérico"}
   467→                  </p>
   468→                </div>
   469→              ))}
   470→            </div>
   471→          </article>
   472→        ))}
   473→      </div>
   474→    </section>
   475→  );
   476→}
