    19→/* ============================================================
    20→   El Camión de las Multiplicaciones — Minijuego educativo
    21→   ------------------------------------------------------------
    22→   Mecánica:
    23→   1. Un cliente del mercado pide N cajitas con M ítems cada una.
    24→   2. El niño arrastra (o toca) N cajitas desde la pila al camión.
    25→   3. Cada cajita muestra visualmente M ítems (emoji) en su interior.
    26→   4. Al presionar "¡Despachar camión!" se valida:
    27→      - Si cargó exactamente N cajitas → celebración + operación formal.
    28→      - Si cargó de más o de menos → feedback pedagógico.
    29→   5. Avance por 10 niveles hasta "¡Transportista Experto!" 🏆.
    30→   ============================================================ */

    76→const SIZE_MAP: Record<CajitaSize, { box: string; emoji: string }> = {
    77→  lg: { box: "w-16 h-16 sm:w-20 sm:h-20 p-1 gap-0.5", emoji: "text-sm sm:text-base" },

    81→
    82→function Cajita({
    83→  producto,
    84→  cantidad,
    85→  cargada,
    86→  draggable,
    87→  onClick,
    88→  onDragStart,
    89→  size = "md",
    90→  index,
    91→}: {
    92→  producto: ProductoVisual;
    93→  cantidad: number;
    94→  cargada?: boolean;
    95→  draggable?: boolean;
    96→  onClick?: () => void;
    97→  onDragStart?: () => void;
    98→  size?: CajitaSize;
    99→  index: number;
   100→}) {
   101→  const emoji = PRODUCTO_EMOJI[producto];
   102→  const cols = cantidad <= 3 ? cantidad : cantidad <= 6 ? 3 : 4;
   103→  const sz = SIZE_MAP[size];
   104→
   105→  return (
   106→    <div
   107→      role={draggable ? "button" : undefined}
   108→      tabIndex={draggable ? 0 : undefined}
   109→      onClick={draggable ? onClick : undefined}
   110→      onKeyDown={(e) => {
   111→        if (!draggable) return;
   112→        if (e.key === "Enter" || e.key === " ") {
   113→          e.preventDefault();
   114→          onClick?.();
   115→        }
   116→      }}
   117→      onDragStart={(e) => {
   118→        if (!draggable) {
   119→          e.preventDefault();
   120→          return;
   121→        }
   122→        e.dataTransfer.setData("text/plain", String(index));
   123→        e.dataTransfer.effectAllowed = "move";
   124→        onDragStart?.();
   125→      }}
   126→      {...(draggable ? ({ draggable: true } as Record<string, unknown>) : {})}
   127→      className={`relative grid shrink-0 select-none place-items-center rounded-lg border-2 shadow-sm transition ${sz.box} ${
   128→        cargada
   129→          ? "border-emerald-500 bg-emerald-50"
   130→          : "border-amber-700 bg-amber-100 hover:border-orange-500 hover:bg-amber-200"
   131→      } ${draggable ? "cursor-grab active:cursor-grabbing hover:scale-105" : ""}`}
   132→      style={{ gridTemplateColumns: `repeat(${cols}, 1fr)` }}
   133→      aria-label={`Cajita ${index + 1} con ${cantidad} ${PRODUCTO_NOMBRE[producto]}`}
   134→      title={`Cajita con ${cantidad} ${PRODUCTO_NOMBRE[producto]}`}
   135→    >
   136→      {Array.from({ length: cantidad }).map((_, i) => (
   137→        <span key={i} className={`${sz.emoji} leading-none flex items-center justify-center`}>
   138→          {emoji}
   139→        </span>
   140→      ))}
   141→    </div>
   142→  );
   143→}

   281→/* ----------------------- Componente principal ----------------------- */
   282→
   283→export function CamionMultiplicaciones() {
   284→  const [nivelIdx, setNivelIdx] = useState(0);
   285→  const [estado, setEstado] = useState<Estado>("presentacion");
   286→  const [cajitasCargadas, setCajitasCargadas] = useState<number[]>([]);
   287→  const [camionesDespachados, setCamionesDespachados] = useState(0);
   288→  const [feedback, setFeedback] = useState<string | null>(null);
   289→  const [dragOverCamion, setDragOverCamion] = useState(false);
   290→
   291→  // Ref para timeouts internos (limpieza segura al desmontar)
   292→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   293→  const programar = useCallback((fn: () => void, ms: number) => {
   294→    const t = setTimeout(() => {
   295→      timeoutsRef.current.delete(t);
   296→      fn();
   297→    }, ms);
   298→    timeoutsRef.current.add(t);
   299→  }, []);
   300→
   301→  useEffect(() => {
   302→    const set = timeoutsRef.current;
   303→    return () => {
   304→      set.forEach((t) => clearTimeout(t));
   305→      set.clear();
   306→    };
   307→  }, []);
   308→
   309→  const nivel = nivelesCamion[nivelIdx];
   310→  const avatar = AVATARES_CLIENTES[nivelIdx % AVATARES_CLIENTES.length];
   311→  const totalPila = nivel.grupos + EXTRAS_PILA;
   312→  const indicesPila = useMemo(
   313→    () => Array.from({ length: totalPila }, (_, i) => i),
   314→    [totalPila],
   315→  );
   316→  const cajitasEnPila = indicesPila.filter((i) => !cajitasCargadas.includes(i));
   317→
   318→  /* ---- Transición: celebrando → siguiente nivel o completado ---- */
   319→  useEffect(() => {
   320→    if (estado !== "celebrando") return;
   321→    const t = setTimeout(() => {
   322→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   323→        setEstado("completado");
   324→      } else {
   325→        setNivelIdx((n) => n + 1);
   326→        setCajitasCargadas([]);
   327→        setFeedback(null);
   328→        setEstado("presentacion");
   329→      }
   330→    }, 2800);
   331→    return () => clearTimeout(t);
   332→  }, [estado, nivelIdx]);
   333→
   334→  /* ---- Acciones ---- */
   335→
   336→  const empezarCarga = useCallback(() => {
   337→    setEstado("cargando");
   338→    setFeedback(null);
   339→  }, []);
   340→
   341→  const cargarCajita = useCallback(
   342→    (i: number) => {
   343→      if (cajitasCargadas.includes(i)) return;
   344→      setCajitasCargadas((p) => [...p, i]);
   345→      setFeedback(null);
   346→    },
   347→    [cajitasCargadas],
   348→  );
   349→
   350→  const quitarCajita = useCallback((i: number) => {
   351→    setCajitasCargadas((p) => p.filter((x) => x !== i));
   352→    setFeedback(null);
   353→  }, []);
   354→
   355→  const vaciarCamion = useCallback(() => {
   356→    setCajitasCargadas([]);
   357→    setFeedback(null);
   358→  }, []);
   359→
   360→  const despachar = useCallback(() => {
   361→    if (estado !== "cargando") return;
   362→    setEstado("verificando");
   363→    const correcto = cajitasCargadas.length === nivel.grupos;
   364→    programar(() => {
   365→      if (correcto) {
   366→        setCamionesDespachados((v) => v + 1);
   367→        setEstado("celebrando");
   368→      } else {
   369→        setFeedback(nivel.feedback_error);
   370→        setEstado("cargando");
   371→      }
   372→    }, 450);
   373→  }, [estado, cajitasCargadas.length, nivel.grupos, nivel.feedback_error, programar]);
   374→
   375→  const reiniciarJuego = useCallback(() => {
   376→    setNivelIdx(0);
   377→    setEstado("presentacion");
   378→    setCajitasCargadas([]);
   379→    setCamionesDespachados(0);
   380→    setFeedback(null);
   381→  }, []);
   382→
   383→  /* ---- Drag handlers para el camión ---- */
   384→  const onDropCamion = useCallback(
   385→    (e: React.DragEvent) => {
   386→      e.preventDefault();
   387→      setDragOverCamion(false);
   388→      const idx = Number(e.dataTransfer.getData("text/plain"));
   389→      if (!Number.isNaN(idx)) cargarCajita(idx);
   390→    },
   391→    [cargarCajita],
   392→  );
   393→
   394→  /* ---- Derivados ---- */
   395→  const progresoNivel = useMemo(
   396→    () => ((nivelIdx + (estado === "celebrando" ? 1 : 0)) / TOTAL_NIVELES) * 100,
   397→    [nivelIdx, estado],
   398→  );
   399→  const esPresentacion = estado === "presentacion";
   400→  const esCargando = estado === "cargando";
   401→  const esVerificando = estado === "verificando";
   402→  const esCelebrando = estado === "celebrando";
   403→  const puedeDespachar = esCargando && cajitasCargadas.length > 0;
   404→
   405→  /* ---- Tamaños de cajitas según cantidad de grupos ---- */
   406→  const sizePila: CajitaSize =
   407→    nivel.grupos > 12 ? "sm" : nivel.grupos > 6 ? "md" : "lg";
   408→  const sizeCamion: CajitaSize = nivel.grupos > 6 ? "sm" : "md";
   409→
   410→  /* ===================== RENDER ===================== */
   411→
   412→  return (
   413→    <div className="relative min-h-[calc(100vh-72px)] overflow-hidden bg-gradient-to-br from-amber-100 via-orange-200 to-rose-200 font-display">
   414→      {/* Toldos de mercado arriba */}
   415→      <ToldosMercado />
   416→
   417→      {/* Decoración de fondo: mercado nicaragüense */}
   418→      <div aria-hidden className="pointer-events-none absolute inset-0 opacity-[0.07]">
   419→        <div className="absolute left-4 top-24 text-7xl">🥭</div>
   420→        <div className="absolute right-8 top-32 text-7xl">🌽</div>
   421→        <div className="absolute left-10 bottom-24 text-7xl">🪅</div>
   422→        <div className="absolute right-12 bottom-32 text-7xl">🥖</div>
   423→        <div className="absolute left-1/3 top-12 text-6xl">☕</div>
   424→        <div className="absolute right-1/3 bottom-16 text-6xl">🍫</div>
   425→      </div>
   426→
   427→      <div className="relative z-10 mx-auto max-w-5xl px-3 pb-6 pt-14 sm:px-5 sm:pt-16">
   428→        {/* ===================== HUD superior ===================== */}
   429→        <div className="mb-4 rounded-2xl border-2 border-teal-200 bg-white/70 p-3 shadow-md backdrop-blur sm:p-4">
   430→          <div className="flex items-center justify-between gap-3">
   431→            <div className="flex items-center gap-2">
   432→              <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-teal-500 to-emerald-500 text-white shadow sm:h-11 sm:w-11">
   433→                <TruckIcon size={20} strokeWidth={2.5} />
   434→              </div>
   435→              <div>
   436→                <p className="text-xs font-bold uppercase tracking-wide text-teal-700/70">
   437→                  El Camión
   438→                </p>
   439→                <p className="text-base font-bold text-teal-950 sm:text-lg">
   440→                  Nivel {nivel.nivel} <span className="text-teal-600/60">de {TOTAL_NIVELES}</span>
   441→                </p>
   442→              </div>
   443→            </div>
   444→            <div className="flex items-center gap-2">
   445→              <div className="flex items-center gap-1.5 rounded-full border-2 border-lime-300 bg-lime-50 px-3 py-1.5">
   446→                <Package size={15} className="text-lime-700" strokeWidth={2.5} />
   447→                <span className="text-sm font-bold text-lime-800">
   448→                  {camionesDespachados} <span className="hidden sm:inline">despachados</span>
   449→                </span>
   450→              </div>
   451→            </div>
   452→          </div>
   453→          {/* Barra de progreso */}
   454→          <div className="mt-2.5 h-2.5 w-full overflow-hidden rounded-full bg-teal-200/70">
   455→            <div
   456→              className="h-full rounded-full bg-gradient-to-r from-teal-400 via-emerald-400 to-lime-400 transition-all duration-700 ease-out"
   457→              style={{ width: `${progresoNivel}%` }}
   458→            />
   459→          </div>
   460→        </div>
   461→
   462→        {/* ===================== Pantalla: completado ===================== */}
   463→        {estado === "completado" ? (
   464→          <PantallaCompletado onReiniciar={reiniciarJuego} camiones={camionesDespachados} />
   465→        ) : (
   466→          <div className="grid gap-4 lg:grid-cols-2">
   467→            {/* ============ Columna izquierda: cliente + camión ============ */}
   468→            <div className="flex flex-col gap-4">
   469→              {/* Cliente con bocadillo */}
   470→              <div className="rounded-3xl border-2 border-rose-200 bg-white/80 p-4 shadow-lg backdrop-blur sm:p-5">
   471→                <div className="flex items-start gap-3">
   472→                  <div className="flex h-16 w-16 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-100 to-rose-100 text-4xl shadow-inner sm:h-20 sm:w-20 sm:text-5xl">
   473→                    {avatar}
   474→                  </div>
   475→                  <div className="relative flex-1 rounded-2xl border-2 border-teal-300 bg-amber-50 px-4 py-3 shadow-sm">
   476→                    <div className="absolute -left-2 top-5 h-4 w-4 rotate-45 border-b-2 border-l-2 border-teal-300 bg-amber-50" />
   477→                    <p className="text-base font-semibold leading-snug text-teal-950 sm:text-lg">
   478→                      {nivel.frase_del_cliente}
   479→                    </p>
   480→                  </div>
   481→                </div>
   482→
   483→                {/* Encargo visual (sin revelar la respuesta) */}
   484→                <div className="mt-3 flex flex-wrap items-center justify-center gap-2 rounded-xl bg-teal-100/60 py-2">
   485→                  <span className="text-sm font-semibold text-teal-800">Encargo:</span>
   486→                  <span className="inline-flex items-center gap-1 rounded-lg bg-white px-3 py-1 text-base font-bold text-teal-900 shadow-sm">
   487→                    {nivel.grupos} cajitas
   488→                  </span>
   489→                  <span className="text-sm font-bold text-teal-500">×</span>
   490→                  <span className="inline-flex items-center gap-1 rounded-lg bg-white px-3 py-1 text-base font-bold text-teal-900 shadow-sm">

   494→
   495→                <p className="mt-2 text-center text-xs font-medium text-teal-700/70">

   516→                }`}
   517→              >
   518→                <div className="mb-2 flex items-center justify-between">

   524→                    className={`rounded-full px-2.5 py-0.5 text-xs font-bold ${

   580→                  </div>

   582→
   583→                {cajitasCargadas.length > 0 && esCargando && (
   584→                  <p className="mt-1.5 text-center text-xs font-semibold text-teal-600/70">

   586→                  </p>

   597→                    <Package size={16} className="text-amber-700" strokeWidth={2.5} />
   598→                    Pila de cajitas
   599→                  </h3>
   600→                  <span className="rounded-full bg-amber-100 px-2.5 py-0.5 text-xs font-bold text-amber-800">
   601→                    {cajitasEnPila.length} disponibles
   602→                  </span>

   619→                    {cajitasEnPila.length === 0 ? (
   620→                      <p className="py-4 text-center text-sm font-semibold text-amber-700/70">
   621→                        Todas las cajitas están en el camión
   622→                      </p>
   623→                    ) : (
   624→                      cajitasEnPila.map((i) => (

   693→                    disabled
   694→                    className="flex flex-1 items-center justify-center gap-2 rounded-2xl bg-gradient-to-r from-lime-400 to-emerald-500 py-3.5 text-base font-bold text-white shadow-md"

   700→            </div>
