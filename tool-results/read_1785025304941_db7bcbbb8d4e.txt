   192→      cancelAnimationFrame(rafId);

   208→      let configActual = configOptimista;

   210→      let gemasActual = data.gemas;
   211→      let ownedActual = [...data.itemsOwned];

   213→      let equipados = 0;

   215→        const item = tienda.find((it) => it.categoria === cat && it.clave === clave);
   216→        if (!item) { faltaron++; continue; }
   217→        const yaPosee = ownedActual.includes(item.id);
   218→        const esGratis = item.precioMonedas === 0 && item.precioGemas === 0;
   219→        if (!esGratis && !yaPosee) {
   220→          if (!isDemo) {
   221→            const nivelOk = (data.nivel) >= item.nivelRequerido;
   222→            const fondosOk = item.precioGemas > 0 ? gemasActual >= item.precioGemas : monedasActual >= item.precioMonedas;
   223→            if (!nivelOk || !fondosOk) { faltaron++; continue; }
   224→          }
   225→          try {
   226→            const res = await api.comprar(usuario.id, item.id);
   227→            monedasActual = res.monedas;
   228→            gemasActual = res.gemas;
   229→            ownedActual = [...ownedActual, item.id];
   230→          } catch {
   231→            faltaron++; continue;
   232→          }
   233→        }
   234→        try {
   235→          const res = await api.equipar(usuario.id, item.id);
   236→          configActual = res.config;
   237→          equipados++;
   238→        } catch {
   239→          faltaron++;
   240→        }
   241→      }
   242→      setData((prev) => (prev ? { ...prev, config: configActual, monedas: monedasActual, gemas: gemasActual, itemsOwned: ownedActual } : prev));
   243→      setUsuario({ ...usuario, monedas: monedasActual, gemas: gemasActual });
   244→      if (faltaron === 0) {
   245→        mostrarToast(`${p.nombre} aplicado · ${equipados} piezas equipadas`, "exito");
   246→      } else if (equipados > 0) {
   247→        mostrarToast(`${p.nombre} parcial · ${equipados} piezas, ${faltaron} bloqueadas`, "info");
   248→      } else {
   249→        mostrarToast(`No se pudo aplicar ${p.nombre}`, "error");
   250→      }
   251→    } catch (err) {
   252→      mostrarToast(err instanceof Error ? err.message : "Error al aplicar el personaje", "error");
   253→      await cargar();
   254→    } finally {
   255→      setAplicandoPersonaje(false);
   256→    }

   277→          if (!isDemo) {
   278→            const nivelOk = (data.nivel) >= item.nivelRequerido;
   279→            const fondosOk = item.precioGemas > 0 ? gemasActual >= item.precioGemas : monedasActual >= item.precioMonedas;

   400→      {/* ===== Decoración de fondo: cuadrícula isométrica ===== */}
   401→      <div aria-hidden className="pointer-events-none absolute inset-0 z-0 overflow-hidden">
