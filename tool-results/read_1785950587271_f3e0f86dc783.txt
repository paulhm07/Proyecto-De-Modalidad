    52→    52→// Quita signos de puntuación de los extremos de un token.
    53→    53→function limpiar(token: string): string {
    54→    54→  return token
    55→    55→    .replace(/^[.,;:!¡¿()"']+/, "")
    56→    56→    .replace(/[.,;:!¡¿()"']+$/, "");
    57→    57→}
    58→    58→
    59→    59→// Normaliza una frase para comparar (minúsculas, sin puntuación, sin dobles espacios).
    60→    60→function normalizar(s: string): string {
    61→    61→  return s
    62→    62→    .toLowerCase()
    63→    63→    .trim()
    64→    64→    .split(/\s+/)
    65→    65→    .map(limpiar)
    66→    66→    .filter(Boolean)
    67→    67→    .join(" ");
    68→    68→}

    72→    72→  return tipo === "pronombre_el_ella" || tipo === "pronombre_ellos_ellas";

    75→    75→// Tokeniza un enunciado en palabras (limpiando puntuación de los extremos).
    76→    76→function tokenizarEnunciado(enunciado: string): string[] {
    77→    77→  return enunciado
    78→    78→    .split(/\s+/)
    79→    79→    .map((t) => limpiar(t))
    80→    80→    .filter(Boolean);
    81→    81→}

    83→    83→// Fisher–Yates sin mutar el original.
    84→    84→function mezclar<T>(arr: readonly T[]): T[] {
    85→    85→  const a = [...arr];
    86→    86→  for (let i = a.length - 1; i > 0; i--) {
    87→    87→    const j = Math.floor(Math.random() * (i + 1));
    88→    88→    [a[i], a[j]] = [a[j], a[i]];
    89→    89→  }
    90→    90→  return a;
    91→    91→}

   259→   259→
   260→   260→export function BusLetras() {
   261→   261→  const [nivelIdx, setNivelIdx] = useState(0);
   262→   262→  const [estado, setEstado] = useState<Estado>("presentacion");
   263→   263→  const [seleccionadas, setSeleccionadas] = useState<number[]>([]);
   264→   264→  const [pasajerosTransportados, setPasajerosTransportados] = useState(0);
   265→   265→  const [feedback, setFeedback] = useState<string | null>(null);
   266→   266→
   267→   267→  // Ref para timeouts internos (limpieza segura al desmontar)
   268→   268→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   269→   269→  const programar = useCallback((fn: () => void, ms: number) => {
   270→   270→    const t = setTimeout(() => {
   271→   271→      timeoutsRef.current.delete(t);
   272→   272→      fn();
   273→   273→    }, ms);
   274→   274→    timeoutsRef.current.add(t);
   275→   275→  }, []);
   276→   276→
   277→   277→  useEffect(() => {
   278→   278→    const set = timeoutsRef.current;
   279→   279→    return () => {
   280→   280→      set.forEach((t) => clearTimeout(t));
   281→   281→      set.clear();
   282→   282→    };
   283→   283→  }, []);
   284→   284→
   285→   285→  const nivel = nivelesBus[nivelIdx];
   286→   286→  const esPronombres = esNivelPronombres(nivel.tipo_ejercicio);
   287→   287→
   288→   288→  // Tokens del enunciado (memo por nivel)
   289→   289→  const tokensEnunciado = useMemo(
   290→   290→    () => tokenizarEnunciado(nivel.enunciado),
   291→   291→    [nivel],
   292→   292→  );
   293→   293→
   294→   294→  // Opciones de pronombres mezcladas (memo por nivel)
   295→   295→  const opcionesPronombres = useMemo(
   296→   296→    () =>
   297→   297→      esPronombres
   298→   298→        ? mezclar([nivel.palabra_correcta, ...nivel.distractores])
   299→   299→        : [],
   300→   300→    // eslint-disable-next-line react-hooks/exhaustive-deps
   301→   301→    [nivel.nivel],
   302→   302→  );
   303→   303→
   304→   304→  /* ---- Transición: celebrando → siguiente nivel o completado ---- */
   305→   305→  useEffect(() => {
   306→   306→    if (estado !== "celebrando") return;
   307→   307→    const t = setTimeout(() => {
   308→   308→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   309→   309→        setEstado("completado");
   310→   310→      } else {
   311→   311→        setNivelIdx((n) => n + 1);
   312→   312→        setSeleccionadas([]);
   313→   313→        setFeedback(null);
   314→   314→        setEstado("presentacion");
   315→   315→      }
   316→   316→    }, 2800);
   317→   317→    return () => clearTimeout(t);
   318→   318→  }, [estado, nivelIdx]);
   319→   319→
   320→   320→  /* ---- Validación ---- */
   321→   321→  const validar = useCallback((): boolean => {
   322→   322→    if (seleccionadas.length === 0) return false;
   323→   323→    if (esPronombres) {
   324→   324→      if (seleccionadas.length !== 1) return false;
   325→   325→      return (
   326→   326→        normalizar(opcionesPronombres[seleccionadas[0]]) ===
   327→   327→        normalizar(nivel.palabra_correcta)
   328→   328→      );
   329→   329→    }
   330→   330→    // Modo enunciado: ordenar índices asc y unir tokens
   331→   331→    const frase = [...seleccionadas]
   332→   332→      .sort((a, b) => a - b)
   333→   333→      .map((i) => tokensEnunciado[i])
   334→   334→      .join(" ");
   335→   335→    return normalizar(frase) === normalizar(nivel.palabra_correcta);
   336→   336→  }, [seleccionadas, esPronombres, opcionesPronombres, nivel.palabra_correcta, tokensEnunciado]);
   337→   337→
   338→   338→  /* ---- Acciones ---- */
   339→   339→  const empezar = useCallback(() => {
   340→   340→    setEstado("seleccionando");
   341→   341→    setFeedback(null);
   342→   342→  }, []);
   343→   343→
   344→   344→  const toggleSeleccion = useCallback(
   345→   345→    (i: number) => {
   346→   346→      if (estado !== "seleccionando") return;
   347→   347→      setFeedback(null);
   348→   348→      if (esPronombres) {
   349→   349→        // En pronombres solo se permite 1 selección a la vez
   350→   350→        setSeleccionadas((p) => (p.includes(i) ? [] : [i]));
   351→   351→        return;
   352→   352→      }
   353→   353→      setSeleccionadas((p) =>
   354→   354→        p.includes(i) ? p.filter((x) => x !== i) : [...p, i],
   355→   355→      );
   356→   356→    },
   357→   357→    [estado, esPronombres],
   358→   358→  );
   359→   359→
   360→   360→  const subirAlBus = useCallback(() => {
   361→   361→    if (estado !== "seleccionando") return;
   362→   362→    if (seleccionadas.length === 0) return;
   363→   363→    setEstado("verificando");
   364→   364→    const correcto = validar();
   365→   365→    programar(() => {
   366→   366→      if (correcto) {
   367→   367→        setPasajerosTransportados((v) => v + 1);
   368→   368→        setEstado("celebrando");
   369→   369→      } else {
   370→   370→        setFeedback(nivel.feedback_error);
   371→   371→        setEstado("seleccionando");
   372→   372→      }
   373→   373→    }, 550);
   374→   374→  }, [estado, seleccionadas.length, validar, nivel.feedback_error, programar]);
   375→   375→
   376→   376→  const limpiarSeleccion = useCallback(() => {
   377→   377→    setSeleccionadas([]);
   378→   378→    setFeedback(null);
   379→   379→  }, []);
   380→   380→
   381→   381→  const reiniciarJuego = useCallback(() => {
   382→   382→    setNivelIdx(0);
   383→   383→    setEstado("presentacion");
   384→   384→    setSeleccionadas([]);
   385→   385→    setPasajerosTransportados(0);
   386→   386→    setFeedback(null);
   387→   387→  }, []);
   388→   388→
   389→   389→  /* ---- Derivados ---- */
   390→   390→  const progresoNivel = useMemo(
   391→   391→    () => ((nivelIdx + (estado === "celebrando" ? 1 : 0)) / TOTAL_NIVELES) * 100,
   392→   392→    [nivelIdx, estado],
   393→   393→  );
   394→   394→  const esPresentacion = estado === "presentacion";
   395→   395→  const esSeleccionando = estado === "seleccionando";
   396→   396→  const esVerificando = estado === "verificando";
   397→   397→  const esCelebrando = estado === "celebrando";
   398→   398→  const puedeSubir = esSeleccionando && seleccionadas.length > 0;
   399→   399→
   400→   400→  /* ===================== RENDER ===================== */
   401→   401→
   402→   402→  return (
   403→   403→    <div className="relative min-h-[calc(100vh-72px)] overflow-hidden bg-gradient-to-br from-rose-100 via-amber-100 to-rose-200 font-display">
   404→   404→      {/* Cielo: sol y nubes */}
   405→   405→      <div aria-hidden className="pointer-events-none absolute inset-0">
   406→   406→        <div className="absolute right-6 top-8 h-16 w-16 rounded-full bg-amber-300 opacity-80 blur-sm sm:right-10 sm:top-12 sm:h-20 sm:w-20" />
   407→   407→        <div className="absolute right-12 top-10 h-12 w-12 rounded-full bg-amber-200 opacity-70 sm:right-16 sm:top-16 sm:h-14 sm:w-14" />
   408→   408→        <div className="absolute left-1/4 top-6 h-6 w-16 rounded-full bg-white/70 blur-[1px]" />
   409→   409→        <div className="absolute right-1/4 top-14 h-5 w-14 rounded-full bg-white/60 blur-[1px]" />
   410→   410→      </div>
   411→   411→
   412→   412→      {/* Palmeras a los lados (solo desktop) */}
   413→   413→      <Palmera className="pointer-events-none absolute -left-2 bottom-2 z-0 hidden h-56 w-32 -rotate-6 opacity-90 sm:block lg:h-72 lg:w-40" />
   414→   414→      <Palmera className="pointer-events-none absolute -right-2 bottom-2 z-0 hidden h-56 w-32 rotate-6 opacity-90 sm:block lg:h-72 lg:w-40" />
   415→   415→
   416→   416→      <div className="relative z-10 mx-auto max-w-5xl px-3 pb-6 pt-4 sm:px-5 sm:pt-6">
   417→   417→        {/* ===================== HUD superior ===================== */}
   418→   418→        <div className="mb-4 rounded-2xl border-2 border-rose-200 bg-white/70 p-3 shadow-md backdrop-blur sm:p-4">
   419→   419→          <div className="flex items-center justify-between gap-3">
   420→   420→            <div className="flex items-center gap-2">
   421→   421→              <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-rose-500 to-fuchsia-500 text-white shadow sm:h-11 sm:w-11">
   422→   422→                <BusIcon size={20} strokeWidth={2.5} />
   423→   423→              </div>
   424→   424→              <div>
   425→   425→                <p className="text-xs font-bold uppercase tracking-wide text-rose-700/70">
   426→   426→                  El Bus de las Letras
   427→   427→                </p>
   428→   428→                <p className="text-base font-bold text-rose-950 sm:text-lg">
   429→   429→                  Nivel {nivel.nivel}{" "}
   430→   430→                  <span className="text-rose-600/60">de {TOTAL_NIVELES}</span>
   431→   431→                </p>
   432→   432→              </div>
