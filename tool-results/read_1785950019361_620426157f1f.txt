   398→   398→interface CartaState {

   403→   403→
   404→   404→export function CartaOrtografia() {
   405→   405→  const [nivelIdx, setNivelIdx] = useState(0);
   406→   406→  const [estado, setEstado] = useState<Estado>("presentacion");
   407→   407→  const [cartasNivel, setCartasNivel] = useState<CartaState[]>([]);
   408→   408→  const [idxActiva, setIdxActiva] = useState<number | null>(null);
   409→   409→  const [feedback, setFeedback] = useState<string | null>(null);
   410→   410→  const [cartasEnviadasTotal, setCartasEnviadasTotal] = useState(0);
   411→   411→
   412→   412→  // Ref para timeouts internos (limpieza segura al desmontar)
   413→   413→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   414→   414→  const programar = useCallback((fn: () => void, ms: number) => {
   415→   415→    const t = setTimeout(() => {
   416→   416→      timeoutsRef.current.delete(t);
   417→   417→      fn();
   418→   418→    }, ms);
   419→   419→    timeoutsRef.current.add(t);
   420→   420→  }, []);
   421→   421→
   422→   422→  useEffect(() => {
   423→   423→    const set = timeoutsRef.current;
   424→   424→    return () => {
   425→   425→      set.forEach((t) => clearTimeout(t));
   426→   426→      set.clear();
   427→   427→    };
   428→   428→  }, []);
   429→   429→
   430→   430→  // Cargar las 4 cartas del nivel actual al cambiar de nivel
   431→   431→  useEffect(() => {
   432→   432→    const cartas = cartasPorNivel(nivelIdx + 1);
   433→   433→    setCartasNivel(
   434→   434→      cartas.map((c) => ({
   435→   435→        carta: c,
   436→   436→        estado: "sin_abrir" as EstadoCarta,
   437→   437→        textoInput: "",
   438→   438→      })),
   439→   439→    );
   440→   440→    setIdxActiva(null);
   441→   441→    setFeedback(null);
   442→   442→  }, [nivelIdx]);
   443→   443→
   444→   444→  const nivel = nivelIdx + 1;
   445→   445→  const selladasNivel = useMemo(
   446→   446→    () => cartasNivel.filter((c) => c.estado === "sellada_correcta").length,
   447→   447→    [cartasNivel],
   448→   448→  );
   449→   449→
   450→   450→  /* ---- Transición: celebrando → siguiente nivel o completado ---- */
   451→   451→  useEffect(() => {
   452→   452→    if (estado !== "celebrando") return;
   453→   453→    const t = setTimeout(() => {
   454→   454→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   455→   455→        setEstado("completado");
   456→   456→      } else {
   457→   457→        setNivelIdx((n) => n + 1);
   458→   458→        setEstado("presentacion");
   459→   459→      }
   460→   460→    }, 2400);
   461→   461→    return () => clearTimeout(t);
   462→   462→  }, [estado, nivelIdx]);
   463→   463→
   464→   464→  // ¿Todas las cartas del nivel están selladas? → celebrar
   465→   465→  useEffect(() => {
   466→   466→    if (estado !== "jugando") return;
   467→   467→    if (cartasNivel.length === 0) return;
   468→   468→    if (selladasNivel === CARTAS_POR_NIVEL) {
   469→   469→      programar(() => setEstado("celebrando"), 400);
   470→   470→    }
   471→   471→  }, [selladasNivel, cartasNivel.length, estado, programar]);
   472→   472→
   473→   473→  /* ---- Acciones ---- */
   474→   474→
   475→   475→  const empezar = useCallback(() => {
   476→   476→    setEstado("jugando");
   477→   477→    setFeedback(null);
   478→   478→  }, []);
   479→   479→
   480→   480→  const abrirCarta = useCallback(
   481→   481→    (i: number) => {
   482→   482→      if (estado !== "jugando") return;
   483→   483→      setCartasNivel((prev) =>
   484→   484→        prev.map((c, idx) => {
   485→   485→          if (idx === i) {
   486→   486→            if (c.estado === "sellada_correcta") return c;
