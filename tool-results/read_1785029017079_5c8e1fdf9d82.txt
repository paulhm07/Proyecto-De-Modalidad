     1→"use client";
     2→
     3→interface PersonajeIconicoSVGProps {
     4→  personajeId: string;
     5→  size?: number;
     6→  className?: string;
     7→}
     8→
     9→/**
    10→ * Renderiza personajes icónicos en estilo bloque 3D Roblox/Minecraft.
    11→ * Cada personaje tiene un cuerpo base humanoide (cabeza cúbica, torso, extremidades)
    12→ * con colores y detalles distintivos.
    13→ *
    14→ * Personajes: Goku, Naruto, Pikachu, Eren, Mario, Master Chief, Sora,
    15→ * Spider-Man, Batman, Luke, Eleven, Mickey.
    16→ */
    17→export function PersonajeIconicoSVG({ personajeId, size = 200, className = "" }: PersonajeIconicoSVGProps) {
    18→  return (
    19→    <svg
    20→      width={size}
    21→      height={size}
    22→      viewBox="0 0 200 240"
    23→      xmlns="http://www.w3.org/2000/svg"
    24→      className={className}
    25→      role="img"
    26→      aria-label={`Personaje ${personajeId}`}
    27→    >
    28→      <defs>
    29→        <radialGradient id="ps-shadow" cx="0.5" cy="0.5" r="0.5">
    30→          <stop offset="0%" stopColor="#000000" stopOpacity="0.35" />
    31→          <stop offset="100%" stopColor="#000000" stopOpacity="0" />
    32→        </radialGradient>
    33→      </defs>
    34→      {/* Sombra del piso */}
    35→      <ellipse cx="100" cy="228" rx="52" ry="9" fill="url(#ps-shadow)" />
    36→      {/* Render del personaje específico */}
    37→      {renderPersonaje(personajeId)}
    38→    </svg>
    39→  );
    40→}
    41→
    42→function renderPersonaje(id: string) {
    43→  switch (id) {
    44→    case "goku": return <GokuSVG />;
    45→    case "naruto": return <NarutoSVG />;
    46→    case "pikachu": return <PikachuSVG />;
    47→    case "eren": return <ErenSVG />;
    48→    case "mario": return <MarioSVG />;
    49→    case "master-chief": return <MasterChiefSVG />;
    50→    case "sora": return <SoraSVG />;
    51→    case "spiderman": return <SpiderManSVG />;
    52→    case "batman": return <BatmanSVG />;
    53→    case "luke": return <LukeSVG />;
    54→    case "eleven": return <ElevenSVG />;
    55→    case "mickey": return <MickeySVG />;
    56→    // Piratas Mugiwara (One Piece)
    57→    case "luffy": return <LuffySVG />;
    58→    case "zoro": return <ZoroSVG />;
    59→    case "nami": return <NamiSVG />;
    60→    case "usopp": return <UsoppSVG />;
    61→    case "sanji": return <SanjiSVG />;
    62→    case "chopper": return <ChopperSVG />;
    63→    case "robin": return <RobinSVG />;
    64→    case "franky": return <FrankySVG />;
    65→    case "brook": return <BrookSVG />;
    66→    case "jinbe": return <JinbeSVG />;
    67→    default: return <GokuSVG />;
    68→  }
    69→}
