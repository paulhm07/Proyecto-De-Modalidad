    64→    64→function polar(cx: number, cy: number, r: number, deg: number) {
    65→    65→  const rad = ((deg - 90) * Math.PI) / 180;
    66→    66→  return { x: cx + r * Math.cos(rad), y: cy + r * Math.sin(rad) };
    67→    67→}
    68→    68→
    69→    69→function pieSlicePath(cx: number, cy: number, r: number, a1: number, a2: number) {
    70→    70→  const p1 = polar(cx, cy, r, a1);
    71→    71→  const p2 = polar(cx, cy, r, a2);
    72→    72→  const largeArc = a2 - a1 > 180 ? 1 : 0;
    73→    73→  return `M ${cx} ${cy} L ${p1.x} ${p1.y} A ${r} ${r} 0 ${largeArc} 1 ${p2.x} ${p2.y} Z`;
    74→    74→}

    98→    98→/** Convierte "Hola $\frac{1}{2}$ porfa" en nodos React con fracción visual. */
    99→    99→function renderFrase(frase: string): React.ReactNode[] {
   100→   100→  const partes: React.ReactNode[] = [];
   101→   101→  const regex = /\$\\frac\{(\d+)\}\{(\d+)\}\$/g;
   102→   102→  let ultimo = 0;
   103→   103→  let m: RegExpExecArray | null;
   104→   104→  let k = 0;
   105→   105→  while ((m = regex.exec(frase)) !== null) {
   106→   106→    if (m.index > ultimo) {
   107→   107→      partes.push(<span key={`t-${k++}`}>{frase.slice(ultimo, m.index)}</span>);
   108→   108→    }
   109→   109→    partes.push(<Fraccion key={`f-${k++}`} num={m[1]} den={m[2]} />);
   110→   110→    ultimo = regex.lastIndex;
   111→   111→  }
   112→   112→  if (ultimo < frase.length) {
   113→   113→    partes.push(<span key={`t-${k++}`}>{frase.slice(ultimo)}</span>);
   114→   114→  }
   115→   115→  return partes;
   116→   116→}

   359→   359→/** Semillas de sandía distribuidas dentro de la pulpa de una rebanada. */
   360→   360→function semillasSandia(indice: number, total: number, a1: number, a2: number) {
   361→   361→  const cx = 100;
   362→   362→  const cy = 100;
   363→   363→  const midAngle = (a1 + a2) / 2;
   364→   364→  const sliceAngle = a2 - a1;
   365→   365→  const numSemillas = total <= 3 ? 3 : total <= 6 ? 2 : 1;
   366→   366→  return Array.from({ length: numSemillas }).map((_, s) => {
   367→   367→    const offset = (s - (numSemillas - 1) / 2) * (sliceAngle / (numSemillas + 1));
   368→   368→    const angle = midAngle + offset;
   369→   369→    const r = 52;
   370→   370→    const pos = polar(cx, cy, r, angle);
   371→   371→    return (
   372→   372→      <ellipse
   373→   373→        key={s}
   374→   374→        cx={pos.x}
   375→   375→        cy={pos.y}
   376→   376→        rx={2.4}
   377→   377→        ry={4}
   378→   378→        fill="#1f2937"
   379→   379→        transform={`rotate(${angle} ${pos.x} ${pos.y})`}
   380→   380→      />
   381→   381→    );
   382→   382→  });

   419→   419→      timeoutsRef.current.delete(t);

   426→   426→  useEffect(() => {

   429→   429→      set.forEach((t) => clearTimeout(t));

   439→   439→  useEffect(() => {

   442→   442→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   443→   443→        setEstado("completado");

   450→   450→        setEstado("presentacion");
   451→   451→      }

   453→   453→    return () => clearTimeout(t);
