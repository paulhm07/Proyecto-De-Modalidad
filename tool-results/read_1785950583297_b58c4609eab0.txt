    28→/* ============================================================
    29→   La Carta Mal Enviada — Minijuego educativo
    30→   ------------------------------------------------------------
    31→   Mecánica:
    32→   1. El niño es "cartero" en la oficina de correos nicaragüense.
    33→      Para cada nivel aparecen 4 sobres.
    34→   2. Toca un sobre para abrirlo y leer el texto_mostrado.
    35→   3. Decide: "✓ Está bien" (la carta se sella y se va al buzón)
    36→      o "✗ Tiene error" (se abre un mini-formulario de corrección).
    37→   4. Si la decisión es correcta:
    38→      - "✓ Está bien" + la carta NO tenía error → se sella con ✓ verde.
    39→      - "✗ Tiene error" + la carta SÍ tenía error → aparece el formulario
    40→        con la corrección_correcta como referencia y la regla_ortografica
    41→        como pista. El niño escribe la corrección y presiona "Corregir".
    42→        Si coincide (case-insensitive, trim) → se sella con ✓ verde.
    43→   5. Si la decisión o la corrección es incorrecta → se muestra
    44→      feedback_error pedagógico y la carta queda abierta para reintentar.
    45→   6. Cuando las 4 cartas del nivel están selladas → confeti y avanza
    46→      al siguiente nivel.
    47→   7. Al completar nivel 10 → "¡Cartero Experto!" 🏆.
    48→   ============================================================ */

    50→type Estado = "presentacion" | "jugando" | "celebrando" | "completado";

    53→  | "sin_abrir"
    54→  | "abierta"
    55→  | "corrigiendo"
    56→  | "sellada_correcta";

    62→/* ----------------------- Helpers ----------------------- */
    63→
    64→// Normaliza una cadena para comparar (minúsculas, sin signos de puntuación
    65→// en los extremos, sin dobles espacios). No toca signos de apertura
    66→// ¡ ¿ ni de cierre ! ? porque son parte de la corrección que se enseña.
    67→function normalizar(s: string): string {
    68→  return s
    69→    .toLowerCase()
    70→    .trim()
    71→    .replace(/\s+/g, " ");
    72→}
    73→
    74→// ¿La carta tiene realmente un error ortográfico?
    75→// Comparamos texto_mostrado contra correccion_correcta (normalizado).
    76→function cartaTieneError(c: Carta): boolean {
    77→  return normalizar(c.texto_mostrado) !== normalizar(c.correccion_correcta);
    78→}

   396→/* ----------------------- Componente principal ----------------------- */
   397→
   398→interface CartaState {
   399→  carta: Carta;
   400→  estado: EstadoCarta;
   401→  textoInput: string;
   402→}
   403→
   404→export function CartaOrtografia() {
   405→  const [nivelIdx, setNivelIdx] = useState(0);
   406→  const [estado, setEstado] = useState<Estado>("presentacion");
   407→  const [cartasNivel, setCartasNivel] = useState<CartaState[]>([]);
   408→  const [idxActiva, setIdxActiva] = useState<number | null>(null);
   409→  const [feedback, setFeedback] = useState<string | null>(null);
   410→  const [cartasEnviadasTotal, setCartasEnviadasTotal] = useState(0);
   411→
   412→  // Ref para timeouts internos (limpieza segura al desmontar)
   413→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   414→  const programar = useCallback((fn: () => void, ms: number) => {
   415→    const t = setTimeout(() => {
   416→      timeoutsRef.current.delete(t);
   417→      fn();
   418→    }, ms);
   419→    timeoutsRef.current.add(t);
   420→  }, []);
   421→
   422→  useEffect(() => {
   423→    const set = timeoutsRef.current;
   424→    return () => {
   425→      set.forEach((t) => clearTimeout(t));
   426→      set.clear();
   427→    };
   428→  }, []);
   429→
   430→  // Cargar las 4 cartas del nivel actual al cambiar de nivel
   431→  useEffect(() => {
   432→    const cartas = cartasPorNivel(nivelIdx + 1);
   433→    setCartasNivel(
   434→      cartas.map((c) => ({
   435→        carta: c,
   436→        estado: "sin_abrir" as EstadoCarta,
   437→        textoInput: "",
   438→      })),
   439→    );
   440→    setIdxActiva(null);
   441→    setFeedback(null);
   442→  }, [nivelIdx]);
   443→
   444→  const nivel = nivelIdx + 1;
   445→  const selladasNivel = useMemo(
   446→    () => cartasNivel.filter((c) => c.estado === "sellada_correcta").length,
   447→    [cartasNivel],
   448→  );
   449→
   450→  /* ---- Transición: celebrando → siguiente nivel o completado ---- */
   451→  useEffect(() => {
   452→    if (estado !== "celebrando") return;
   453→    const t = setTimeout(() => {
   454→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   455→        setEstado("completado");
   456→      } else {
   457→        setNivelIdx((n) => n + 1);
   458→        setEstado("presentacion");
   459→      }
   460→    }, 2400);
   461→    return () => clearTimeout(t);
   462→  }, [estado, nivelIdx]);
   463→
   464→  // ¿Todas las cartas del nivel están selladas? → celebrar
   465→  useEffect(() => {
   466→    if (estado !== "jugando") return;
   467→    if (cartasNivel.length === 0) return;
   468→    if (selladasNivel === CARTAS_POR_NIVEL) {
   469→      programar(() => setEstado("celebrando"), 400);
   470→    }
   471→  }, [selladasNivel, cartasNivel.length, estado, programar]);
   472→
   473→  /* ---- Acciones ---- */
   474→
   475→  const empezar = useCallback(() => {
   476→    setEstado("jugando");
   477→    setFeedback(null);
   478→  }, []);
   479→
   480→  const abrirCarta = useCallback(
   481→    (i: number) => {
   482→      if (estado !== "jugando") return;
   483→      setCartasNivel((prev) =>
   484→        prev.map((c, idx) => {
   485→          if (idx === i) {
   486→            if (c.estado === "sellada_correcta") return c;
   487→            return { ...c, estado: "abierta" as EstadoCarta };
   488→          }
   489→          // Las demás se vuelven "sin_abrir" si estaban abiertas sin decisión
   490→          if (c.estado === "abierta" || c.estado === "corrigiendo") {
   491→            return { ...c, estado: "sin_abrir" as EstadoCarta };
   492→          }
   493→          return c;
   494→        }),
   495→      );
   496→      setIdxActiva(i);
   497→      setFeedback(null);
   498→    },
   499→    [estado],
   500→  );
   501→
   502→  const decidir = useCallback(
   503→    (i: number, diceError: boolean) => {
   504→      if (estado !== "jugando") return;
   505→      const cartaObj = cartasNivel[i]?.carta;
   506→      if (!cartaObj) return;
   507→      const tieneError = cartaTieneError(cartaObj);
   508→
   509→      if (diceError) {
   510→        // El niño dice "✗ Tiene error"
   511→        if (tieneError) {
   512→          // Correcto: abrir formulario de corrección
   513→          setCartasNivel((prev) =>
   514→            prev.map((c, idx) =>
   515→              idx === i
   516→                ? {
   517→                    ...c,
   518→                    estado: "corrigiendo" as EstadoCarta,
   519→                    textoInput: cartaObj.texto_mostrado,
   520→                  }
   521→                : c,
   522→            ),
   523→          );
   524→          setFeedback(null);
   525→        } else {
   526→          // Incorrecto: la carta no tenía error
   527→          setFeedback(cartaObj.feedback_error);
   528→        }
   529→      } else {
   530→        // El niño dice "✓ Está bien"
   531→        if (!tieneError) {
   532→          // Correcto: sellar
   533→          setCartasNivel((prev) =>
   534→            prev.map((c, idx) =>
   535→              idx === i ? { ...c, estado: "sellada_correcta" as EstadoCarta } : c,
   536→            ),
   537→          );
   538→          setCartasEnviadasTotal((v) => v + 1);
   539→          setIdxActiva(null);
   540→          setFeedback(null);
   541→        } else {
   542→          // Incorrecto: la carta sí tenía error
   543→          setFeedback(cartaObj.feedback_error);
   544→        }
   545→      }
   546→    },
   547→    [estado, cartasNivel],
   548→  );
   549→
   550→  const cambiarInput = useCallback((i: number, valor: string) => {
   551→    setCartasNivel((prev) =>
   552→      prev.map((c, idx) => (idx === i ? { ...c, textoInput: valor } : c)),
   553→    );
   554→  }, []);
   555→
   556→  const corregir = useCallback(
   557→    (i: number) => {
   558→      if (estado !== "jugando") return;
   559→      const cs = cartasNivel[i];
   560→      if (!cs) return;
   561→      const ok =
   562→        normalizar(cs.textoInput) === normalizar(cs.carta.correccion_correcta);
   563→      if (ok) {
   564→        setCartasNivel((prev) =>
   565→          prev.map((c, idx) =>
   566→            idx === i ? { ...c, estado: "sellada_correcta" as EstadoCarta } : c,
   567→          ),
   568→        );
   569→        setCartasEnviadasTotal((v) => v + 1);
   570→        setIdxActiva(null);
   571→        setFeedback(null);
   572→      } else {
   573→        setFeedback(cs.carta.feedback_error);
   574→      }
   575→    },
   576→    [estado, cartasNivel],
   577→  );
   578→
   579→  const cancelarCorreccion = useCallback(
   580→    (i: number) => {
   581→      setCartasNivel((prev) =>
   582→        prev.map((c, idx) =>
   583→          idx === i
   584→            ? {
   585→                ...c,
   586→                estado: "abierta" as EstadoCarta,
   587→                textoInput: "",
   588→              }
   589→            : c,
   590→        ),
   591→      );
   592→      setFeedback(null);
   593→    },
   594→    [],
   595→  );
   596→
   597→  const reiniciarJuego = useCallback(() => {
   598→    setNivelIdx(0);
   599→    setEstado("presentacion");
   600→    setCartasEnviadasTotal(0);
   601→    setIdxActiva(null);
   602→    setFeedback(null);

  1004→}
