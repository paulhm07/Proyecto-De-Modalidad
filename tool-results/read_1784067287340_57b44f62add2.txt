   193→   193→
   194→   194→function AsignaturasTab({
   195→   195→  asignaturas,
   196→   196→  usuarioId,
   197→   197→  onChange,
   198→   198→  onSeleccionar,
   199→   199→  mostrarToast,
   200→   200→}: {
   201→   201→  asignaturas: Asignatura[];
   202→   202→  usuarioId: string;
   203→   203→  onChange: () => void;
   204→   204→  onSeleccionar: (id: string) => void;
   205→   205→  mostrarToast: (m: string, t?: "exito" | "error" | "info") => void;
   206→   206→}) {
   207→   207→  const [nombre, setNombre] = useState("");
   208→   208→  const [descripcion, setDescripcion] = useState("");
   209→   209→  const [guardando, setGuardando] = useState(false);
   210→   210→
   211→   211→  const crear = async (e: React.FormEvent) => {
   212→   212→    e.preventDefault();
   213→   213→    if (!nombre.trim()) {
   214→   214→      mostrarToast("Escribe un nombre", "error");
   215→   215→      return;
   216→   216→    }
   217→   217→    setGuardando(true);
   218→   218→    try {
   219→   219→      await api.crearAsignatura(nombre.trim(), descripcion.trim() || null, usuarioId);
   220→   220→      mostrarToast("¡Asignatura creada!", "exito");
   221→   221→      setNombre("");
   222→   222→      setDescripcion("");
   223→   223→      onChange();
   224→   224→    } catch (err) {
   225→   225→      mostrarToast(err instanceof Error ? err.message : "Error al crear asignatura", "error");
   226→   226→    } finally {
   227→   227→      setGuardando(false);
   228→   228→    }
   229→   229→  };
   230→   230→
   231→   231→  const eliminar = async (id: string) => {
   232→   232→    if (!confirm("¿Eliminar esta asignatura? Se borrarán sus módulos y desafíos.")) return;
   233→   233→    try {
   234→   234→      await api.eliminarAsignatura(id);
   235→   235→      mostrarToast("Asignatura eliminada", "info");
   236→   236→      onChange();
   237→   237→    } catch (err) {
   238→   238→      mostrarToast(err instanceof Error ? err.message : "Error al eliminar", "error");
   239→   239→    }
   240→   240→  };
   241→   241→
   242→   242→  return (
   243→   243→    <div className="grid gap-5 lg:grid-cols-[1fr_1.4fr]">
   244→   244→      <form onSubmit={crear} className="card-premium rounded-3xl p-5">
   245→   245→        <div className="mb-3 flex items-center gap-2">
   246→   246→          <Plus size={16} className="text-fuchsia-500" strokeWidth={2.5} />
   247→   247→          <h2 className="text-base font-black text-stone-800">Nueva asignatura</h2>
   248→   248→        </div>
   249→   249→        <div className="space-y-3">
   250→   250→          <div>
   251→   251→            <label className="mb-1 block text-xs font-bold text-stone-600">Nombre *</label>
   252→   252→            <input
   253→   253→              type="text"
   254→   254→              value={nombre}
   255→   255→              onChange={(e) => setNombre(e.target.value)}
   256→   256→              placeholder="Ej: Matemáticas"
   257→   257→              maxLength={80}
   258→   258→              className="w-full rounded-2xl border-2 border-orange-200 bg-white px-4 py-2.5 text-sm font-bold text-stone-800 outline-none transition-all placeholder:font-medium placeholder:text-stone-400 focus:border-orange-400 focus:ring-2 focus:ring-orange-200"
   259→   259→            />
   260→   260→          </div>
   261→   261→          <div>
   262→   262→            <label className="mb-1 block text-xs font-bold text-stone-600">Descripción</label>
   263→   263→            <textarea
   264→   264→              value={descripcion}
   265→   265→              onChange={(e) => setDescripcion(e.target.value)}
   266→   266→              placeholder="Breve descripción de la asignatura"
   267→   267→              rows={3}
   268→   268→              maxLength={240}
   269→   269→              className="w-full resize-none rounded-2xl border-2 border-orange-200 bg-white px-4 py-2.5 text-sm font-medium text-stone-800 outline-none transition-all placeholder:font-medium placeholder:text-stone-400 focus:border-orange-400 focus:ring-2 focus:ring-orange-200"
   270→   270→            />
   271→   271→          </div>
   272→   272→          <button
   273→   273→            type="submit"
   274→   274→            disabled={guardando}
   275→   275→            className="btn-3d flex w-full items-center justify-center gap-1.5 rounded-2xl bg-gradient-to-r from-fuchsia-500 to-rose-500 py-2.5 text-sm font-bold text-white shadow-md transition-all hover:scale-[1.02] disabled:opacity-60"
   276→   276→          >
   277→   277→            {guardando ? <Loader2 size={15} className="animate-spin" /> : <Plus size={15} strokeWidth={2.5} />}
   278→   278→            Crear asignatura
   279→   279→          </button>
   280→   280→        </div>
   281→   281→      </form>
   282→   282→
   283→   283→      <div>
   284→   284→        <div className="mb-3 flex items-center gap-2">
   285→   285→          <BookOpen size={16} className="text-orange-500" strokeWidth={2.5} />
   286→   286→          <h2 className="text-base font-black text-stone-800">
   287→   287→            Asignaturas creadas
   288→   288→          </h2>
   289→   289→          <span className="rounded-full bg-orange-100 px-2 py-0.5 text-xs font-bold text-orange-700">
   290→   290→            {asignaturas.length}
   291→   291→          </span>
   292→   292→        </div>
   293→   293→        {asignaturas.length === 0 ? (
   294→   294→          <div className="card-premium rounded-3xl p-6 text-center">
   295→   295→            <BookOpen size={32} className="mx-auto mb-2 text-orange-300" />
   296→   296→            <p className="text-sm font-bold text-stone-600">Aún no has creado asignaturas.</p>
   297→   297→          </div>
   298→   298→        ) : (
   299→   299→          <div className="space-y-2.5">
   300→   300→            {asignaturas.map((a, i) => (
   301→   301→              <div
   302→   302→                key={a.id}
   303→   303→                className="card-premium animate-pop flex items-center gap-3 rounded-2xl p-3.5"
   304→   304→                style={{ animationDelay: `${i * 50}ms` }}
   305→   305→              >
   306→   306→                <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-fuchsia-400 to-rose-500 text-sm font-black text-white">
   307→   307→                  {a.nombre?.charAt(0)?.toUpperCase() ?? "📚"}
   308→   308→                </div>
   309→   309→                <div className="flex-1 min-w-0">
   310→   310→                  <p className="truncate text-sm font-black text-stone-800">{a.nombre}</p>
   311→   311→                  {a.descripcion && (
   312→   312→                    <p className="truncate text-xs font-medium text-stone-500">{a.descripcion}</p>
   313→   313→                  )}
   314→   314→                  <span className="mt-0.5 inline-block rounded-full bg-orange-100 px-1.5 py-0.5 text-[10px] font-bold text-orange-700">
   315→   315→                    {a._count?.modulos ?? 0} módulos
   316→   316→                  </span>
   317→   317→                </div>
   318→   318→                <button
   319→   319→                  onClick={() => onSeleccionar(a.id)}
   320→   320→                  className="btn-3d rounded-xl bg-gradient-to-r from-emerald-400 to-teal-500 px-3 py-2 text-xs font-bold text-white shadow-md transition-all hover:scale-105"
   321→   321→                >
   322→   322→                  Módulos
   323→   323→                </button>
   324→   324→                <button
   325→   325→                  onClick={() => eliminar(a.id)}
   326→   326→                  className="rounded-xl border-2 border-rose-200 bg-white p-2 text-rose-500 transition-all hover:scale-105 hover:bg-rose-50"
   327→   327→                  aria-label="Eliminar"
   328→   328→                >
   329→   329→                  <Trash2 size={14} strokeWidth={2.5} />
   330→   330→                </button>
   331→   331→              </div>
   332→   332→            ))}
   333→   333→          </div>
   334→   334→        )}
   335→   335→      </div>
   336→   336→    </div>
   337→   337→  );
   338→   338→}

   341→   341→
   342→   342→function ModulosTab({
   343→   343→  asignaturas,
   344→   344→  asignaturaSel,
   345→   345→  setAsignaturaSel,
   346→   346→  modulos,
   347→   347→  onChange,
   348→   348→  onSeleccionar,
   349→   349→  mostrarToast,
   350→   350→}: {
   351→   351→  asignaturas: Asignatura[];
   352→   352→  asignaturaSel: string | null;
   353→   353→  setAsignaturaSel: (id: string | null) => void;
   354→   354→  modulos: Modulo[];
   355→   355→  onChange: () => void;
   356→   356→  onSeleccionar: (id: string) => void;
   357→   357→  mostrarToast: (m: string, t?: "exito" | "error" | "info") => void;
   358→   358→}) {
   359→   359→  const [titulo, setTitulo] = useState("");
   360→   360→  const [nivelMinimo, setNivelMinimo] = useState(1);
   361→   361→  const [guardando, setGuardando] = useState(false);
   362→   362→
   363→   363→  const crear = async (e: React.FormEvent) => {
   364→   364→    e.preventDefault();
   365→   365→    if (!asignaturaSel) {
   366→   366→      mostrarToast("Selecciona una asignatura", "error");
   367→   367→      return;
   368→   368→    }
   369→   369→    if (!titulo.trim()) {
   370→   370→      mostrarToast("Escribe un título para el módulo", "error");
   371→   371→      return;
   372→   372→    }
   373→   373→    setGuardando(true);
   374→   374→    try {
   375→   375→      await api.crearModulo(asignaturaSel, titulo.trim(), nivelMinimo);
   376→   376→      mostrarToast("¡Módulo creado!", "exito");
   377→   377→      setTitulo("");
   378→   378→      setNivelMinimo(1);
   379→   379→      onChange();
   380→   380→    } catch (err) {
   381→   381→      mostrarToast(err instanceof Error ? err.message : "Error al crear módulo", "error");
   382→   382→    } finally {
   383→   383→      setGuardando(false);
   384→   384→    }
   385→   385→  };
   386→   386→
   387→   387→  const eliminar = async (id: string) => {
   388→   388→    if (!confirm("¿Eliminar este módulo y sus desafíos?")) return;
   389→   389→    try {
   390→   390→      await api.eliminarModulo(id);
   391→   391→      mostrarToast("Módulo eliminado", "info");
   392→   392→      onChange();
   393→   393→    } catch (err) {
   394→   394→      mostrarToast(err instanceof Error ? err.message : "Error al eliminar", "error");
   395→   395→    }
   396→   396→  };
   397→   397→
   398→   398→  return (
   399→   399→    <div>
   400→   400→      {/* Selector de asignatura */}
   401→   401→      <div className="card-premium mb-4 rounded-3xl p-4">
   402→   402→        <label className="mb-2 block text-xs font-bold text-stone-600">
   403→   403→          Asignatura
   404→   404→        </label>
   405→   405→        <div className="relative">
   406→   406→          <select
   407→   407→            value={asignaturaSel ?? ""}
   408→   408→            onChange={(e) => setAsignaturaSel(e.target.value || null)}
   409→   409→            className="w-full appearance-none rounded-2xl border-2 border-orange-200 bg-white px-4 py-2.5 pr-10 text-sm font-bold text-stone-800 outline-none transition-all focus:border-orange-400 focus:ring-2 focus:ring-orange-200"
   410→   410→          >
   411→   411→            <option value="">— Selecciona una asignatura —</option>
   412→   412→            {asignaturas.map((a) => (
   413→   413→              <option key={a.id} value={a.id}>
   414→   414→                {a.nombre}
   415→   415→              </option>
   416→   416→            ))}
   417→   417→          </select>
   418→   418→          <ChevronDown
   419→   419→            size={16}
   420→   420→            className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-stone-400"
   421→   421→          />
   422→   422→        </div>
   423→   423→      </div>
   424→   424→
   425→   425→      {!asignaturaSel ? (
   426→   426→        <div className="card-premium rounded-3xl p-8 text-center">
   427→   427→          <Layers size={32} className="mx-auto mb-2 text-orange-300" />
   428→   428→          <p className="text-sm font-bold text-stone-600">
   429→   429→            Selecciona una asignatura para gestionar sus módulos.
   430→   430→          </p>
   431→   431→        </div>
   432→   432→      ) : (
