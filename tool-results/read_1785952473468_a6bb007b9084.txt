    19→/* ============================================================
    20→   Atrapa el Acento — Minijuego educativo de ritmo ortográfico
    21→   ------------------------------------------------------------
    22→   Mecánica:
    23→   1. Aparece una palabra incompleta (silabeada, sin tilde) en
    24→      el centro de la pantalla. La sílaba tónica está resaltada
    25→      con bg-rose-200 + una flecha ↓ que indica "aquí va la tilde".
    26→   2. Tres "burbujas" redondas con vocales con tilde flotan por
    27→      la pantalla rebotando en los bordes (animación rAF tipo
    28→      Pong). Una es la correcta; dos son distractores.
    29→   3. El niño debe tocar la burbuja correcta antes de que se
    30→      acabe el tiempo (5 s por palabra).
    31→   4. Acierto → la burbuja "explota" + 10 puntos, "¡Atrapada! 🎉",
    32→      auto-avance en 1.2 s.
    33→   5. Error → la burbuja tocada tiembla, -3 puntos, se muestra
    34→      la regla ortográfica en un callout ámbar; se permite
    35→      reintentar (no avanza).
    36→   6. Tiempo agotado → se muestra la palabra completa con la
    37→      tilde resaltada en verde, +0 puntos, auto-avance en 1.5 s.
    38→   7. Al completar los 20 desafíos → "¡Atrapa-Tilde Maestro!" 🏆.
    39→   ============================================================ */

    88→/* ----------------------- Helpers de audio ----------------------- */

    90→// Pasa a minúsculas antes de hablar: el campo audio_guia tiene la
    91→// sílaba tónica en MAYÚSCULAS como anotación de diseño, pero
    92→// SpeechSynthesis no interpreta mayúsculas como énfasis.
    93→function normalizarAudioGuia(texto: string): string {
    94→  return texto.toLowerCase();
    95→}
    96→
    97→function hablar(texto: string) {
    98→  if (typeof window === "undefined") return;
    99→  if (!("speechSynthesis" in window)) return;
   100→  try {
   101→    window.speechSynthesis.cancel();
   102→    const u = new SpeechSynthesisUtterance(normalizarAudioGuia(texto));
   103→    u.lang = "es-ES";
   104→    u.rate = 0.85;
   105→    u.pitch = 1.1;
   106→    window.speechSynthesis.speak(u);
   107→  } catch {
   108→    /* noop */
   109→  }
   110→}
   111→
   112→function silenciar() {
   113→  if (typeof window === "undefined") return;
   114→  if (!("speechSynthesis" in window)) return;
   115→  try {
   116→    window.speechSynthesis.cancel();
   117→  } catch {
   118→    /* noop */
   119→  }
   120→}

   122→/* ----------------------- Helpers de layout ----------------------- */
   123→
   124→function slotAX(
   125→  slot: "izquierda" | "centro" | "derecha",
   126→  ancho: number,
   127→  size: number,
   128→): number {
   129→  if (slot === "izquierda") return Math.max(8, ancho * 0.12);
   130→  if (slot === "centro") return Math.max(8, ancho * 0.5 - size / 2);
   131→  return Math.max(8, ancho * 0.88 - size);
   132→}
   133→
   134→function slotBY(slotIdx: number, alto: number, size: number): number {
   135→  // Tres filas virtuales para espaciar las burbujas al iniciar.
   136→  const filas = [alto * 0.18, alto * 0.45, alto * 0.72];
   137→  const f = filas[slotIdx % filas.length];
   138→  return Math.min(Math.max(8, f), Math.max(8, alto - size - 8));
   139→}
   140→
   141→function elegir<T>(arr: T[], n: number): T[] {
   142→  const copia = [...arr];
   143→  for (let i = copia.length - 1; i > 0; i--) {
   144→    const j = Math.floor(Math.random() * (i + 1));
   145→    [copia[i], copia[j]] = [copia[j], copia[i]];
   146→  }
   147→  return copia.slice(0, n);
   148→}
   149→
   150→function shuffleThree<T>(arr: [T, T, T]): [T, T, T] {
   151→  const copia = [...arr];
   152→  for (let i = copia.length - 1; i > 0; i--) {
   153→    const j = Math.floor(Math.random() * (i + 1));
   154→    [copia[i], copia[j]] = [copia[j], copia[i]];
   155→  }
   156→  return [copia[0], copia[1], copia[2]];
   157→}
   158→
   159→function shuffleTwo<T>(arr: [T, T]): [T, T] {
   160→  return Math.random() < 0.5 ? [arr[0], arr[1]] : [arr[1], arr[0]];
   161→}

   163→/* ----------------------- Componente principal ----------------------- */
   164→
   165→export function AtrapaAcento() {
   166→  // --- Estado del juego ---
   167→  const [estado, setEstado] = useState<Estado>("presentacion");
   168→  const [idxDesafio, setIdxDesafio] = useState(0);
   169→  const [puntos, setPuntos] = useState(0);
   170→  const [aciertos, setAciertos] = useState(0);
   171→  const [errores, setErrores] = useState(0);
   172→  const [streak, setStreak] = useState(0);
   173→  const [streakMax, setStreakMax] = useState(0);
   174→  const [tiempoRestante, setTiempoRestante] = useState(TIEMPO_POR_PALABRA);
   175→  const [burbujas, setBurbujas] = useState<Burbuja[]>([]);
   176→  const [colorBurbujas, setColorBurbujas] = useState(PALETA_BURBUJAS[0]);
   177→  const [reglaVisible, setReglaVisible] = useState(false);
   178→  const [idxBurbujaAtrapada, setIdxBurbujaAtrapada] = useState<number | null>(null);
   179→  const [idxBurbujaFallida, setIdxBurbujaFallida] = useState<number | null>(null);
   180→
   181→  // Ref espejo de tiempoRestante para poder leer el valor actual dentro
   182→  // del callback del interval sin reiniciar el effect.
   183→  const tiempoRestanteRef = useRef(TIEMPO_POR_PALABRA);
   184→
   185→  const containerRef = useRef<HTMLDivElement>(null);
   186→  const dimsRef = useRef({ w: 800, h: 420 });
   187→  const [esDesktop, setEsDesktop] = useState(false);
   188→  const bubbleSize = esDesktop ? 96 : 80;
   189→
   190→  // Set de timeouts para limpieza segura al desmontar.
   191→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   192→  const programar = useCallback((fn: () => void, ms: number) => {
   193→    const t = setTimeout(() => {
   194→      timeoutsRef.current.delete(t);
   195→      fn();
   196→    }, ms);
   197→    timeoutsRef.current.add(t);
   198→  }, []);
   199→
   200→  useEffect(() => {
   201→    const set = timeoutsRef.current;
   202→    return () => {
   203→      set.forEach((t) => clearTimeout(t));
   204→      set.clear();
   205→      silenciar();
   206→    };
   207→  }, []);
   208→
   209→  // Mantener dimsRef sincronizado con el tamaño real del contenedor
   210→  // para que el rAF loop siempre lea el valor más reciente.
   211→  useEffect(() => {
   212→    const el = containerRef.current;
   213→    if (!el) return;
   214→    const ro = new ResizeObserver((entries) => {
   215→      const e = entries[0];
   216→      if (!e) return;
   217→      const w = Math.max(200, Math.floor(e.contentRect.width));
   218→      const h = Math.max(200, Math.floor(e.contentRect.height));
   219→      dimsRef.current = { w, h };
   220→    });
   221→    ro.observe(el);
   222→    return () => ro.disconnect();
   223→  }, []);
   224→
   225→  // Detectar breakpoint sm (≥640px) para escoger tamaño de burbuja.
   226→  useEffect(() => {
   227→    if (typeof window === "undefined") return;
   228→    const mq = window.matchMedia("(min-width: 640px)");
   229→    const onChange = () => setEsDesktop(mq.matches);
   230→    onChange();
   231→    mq.addEventListener("change", onChange);
   232→    return () => mq.removeEventListener("change", onChange);
   233→  }, []);
   234→
   235→  const desafio: DesafioAtrapa | null = desafiosAtrapa[idxDesafio] ?? null;
   236→
   237→  /* ----------------------- Carga de un desafío ----------------------- */
   238→
   239→  const cargarDesafio = useCallback(
   240→    (idx: number) => {
   241→      const d = desafiosAtrapa[idx];
   242→      if (!d) return;
   243→      const { w, h } = dimsRef.current;
   244→      const size = window.matchMedia("(min-width: 640px)").matches ? 96 : 80;
   245→
   246→      // Construir las 3 vocales (correcta + 2 distractores) en orden
   247→      // aleatorio, y repartir las 3 slots horizontales (izq/centro/der)
   248→      // entre ellas. La correcta arranca en el slot indicado por
   249→      // d.posicion_x; las otras dos toman los slots restantes.
   250→      const slotsRestantes = (
   251→        ["izquierda", "centro", "derecha"] as const
   252→      ).filter((s) => s !== d.posicion_x);
   253→      const [s1, s2] = shuffleTwo([slotsRestantes[0], slotsRestantes[1]]);
   254→      const vocalesSlots: { vocal: string; esCorrecta: boolean; slot: "izquierda" | "centro" | "derecha" }[] = [
   255→        { vocal: d.letra_con_tilde_correcta, esCorrecta: true, slot: d.posicion_x },
   256→        { vocal: d.distractores[0], esCorrecta: false, slot: s1 },
   257→        { vocal: d.distractores[1], esCorrecta: false, slot: s2 },
   258→      ];
   259→      const orden = shuffleThree([0, 1, 2] as [0, 1, 2]);
   260→      const nuevasBurbujas: Burbuja[] = orden.map((slotIdx, i) => {
   261→        const v = vocalesSlots[slotIdx];
   262→        // Velocidades aleatorias pero direcciones distintas.
   263→        // Magnitud base ~110 px/s, modulada por la velocidad del nivel
   264→        // (aumenta ligeramente cada 5 palabras).
   265→        const boost = 1 + Math.floor(idx / 5) * 0.12;
   266→        const speed = (95 + Math.random() * 45) * boost;
   267→        // Ángulo distinto por índice para que no se muevan en grupo.
   268→        const angulos = [
   269→          (Math.PI / 4) + (Math.random() - 0.5) * 0.6, // ↘
   270→          (3 * Math.PI / 4) + (Math.random() - 0.5) * 0.6, // ↙
   271→          -Math.PI / 4 + (Math.random() - 0.5) * 0.6, // ↗
   272→        ];
   273→        const ang = angulos[i % angulos.length];
   274→        return {
   275→          id: i,
   276→          vocal: v.vocal,
   277→          esCorrecta: v.esCorrecta,
   278→          x: slotAX(v.slot, w, size),
   279→          y: slotBY(i, h, size),
   280→          vx: Math.cos(ang) * speed,
   281→          vy: Math.sin(ang) * speed,
   282→          temblando: false,
   283→        };
   284→      });
   285→
   286→      setBurbujas(nuevasBurbujas);
   287→      setColorBurbujas(elegir(PALETA_BURBUJAS, 1)[0]);
   288→      tiempoRestanteRef.current = TIEMPO_POR_PALABRA;
   289→      setTiempoRestante(TIEMPO_POR_PALABRA);
   290→      setReglaVisible(false);
   291→      setIdxBurbujaAtrapada(null);
   292→      setIdxBurbujaFallida(null);
   293→      setEstado("jugando");
   294→
   295→      // Reproducir automáticamente el audio guía al iniciar.
   296→      hablar(d.audio_guia);
   297→    },
   298→    [],
   299→  );
   300→
   301→  /* ----------------------- Loop de animación de burbujas (rAF) ----------------------- */
   302→
   303→  useEffect(() => {
   304→    if (
   305→      estado !== "jugando" &&
   306→      estado !== "fallando" &&
   307→      estado !== "acertando"
   308→    ) {
   309→      return;
   310→    }
   311→    let raf = 0;
   312→    let last = performance.now();
   313→    const loop = (now: number) => {
   314→      const dt = Math.min(50, now - last) / 1000;
   315→      last = now;
   316→      const { w, h } = dimsRef.current;
   317→      const size = window.matchMedia("(min-width: 640px)").matches ? 96 : 80;
   318→      const maxX = Math.max(0, w - size);
   319→      const maxY = Math.max(0, h - size);
   320→      setBurbujas((prev) =>
   321→        prev.map((b) => {
   322→          if (estado === "acertando" && b.id === idxBurbujaAtrapada) {
   323→            // La burbuja atrapada se queda quieta (luego "explota").
   324→            return b;
   325→          }
   326→          let { x, y, vx, vy } = b;
   327→          x += vx * dt;
   328→          y += vy * dt;
   329→          if (x < 0) {
   330→            x = 0;
   331→            vx = -vx;
   332→          } else if (x > maxX) {
   333→            x = maxX;
   334→            vx = -vx;
   335→          }
   336→          if (y < 0) {
   337→            y = 0;
   338→            vy = -vy;
   339→          } else if (y > maxY) {
   340→            y = maxY;
   341→            vy = -vy;
   342→          }
   343→          return { ...b, x, y, vx, vy };
   344→        }),
   345→      );
   346→      raf = requestAnimationFrame(loop);
   347→    };
   348→    raf = requestAnimationFrame(loop);
   349→    return () => cancelAnimationFrame(raf);
   350→  }, [estado, idxBurbujaAtrapada]);
   351→
   352→  /* ----------------------- Timer (5s por palabra) ----------------------- */
   353→  // Solo corre en 'jugando'. Se pausa durante 'fallando' y 'acertando'.
   354→  // La transición a 'tiempo_agotado' se dispara DENTRO del callback del
   355→  // interval (no en un effect body) para evitar renders en cascada.
   356→
   357→  useEffect(() => {
   358→    if (estado !== "jugando") return;
   359→    const id = setInterval(() => {
   360→      const next = Math.max(0, tiempoRestanteRef.current - 0.1);
   361→      tiempoRestanteRef.current = next;
   362→      setTiempoRestante(next);
   363→      if (next <= 0) {
   364→        // Tiempo agotado: transición directa desde el callback.
   365→        setStreak(0);
   366→        setEstado("tiempo_agotado");
   367→      }
   368→    }, 100);
   369→    return () => clearInterval(id);
   370→  }, [estado]);
   371→
   372→  /* ----------------------- Transiciones de estado ----------------------- */
   373→
   374→  // acertando → siguiente desafío (o completado) tras 1.2s
   375→  useEffect(() => {
   376→    if (estado !== "acertando") return;
   377→    const t = setTimeout(() => {
   378→      if (idxDesafio + 1 >= TOTAL_DESAFIOS) {
   379→        setEstado("completado");
   380→        silenciar();
   381→      } else {
   382→        setIdxDesafio((i) => i + 1);
   383→      }
   384→    }, DURACION_ACERTAR_MS);
   385→    return () => clearTimeout(t);
   386→  }, [estado, idxDesafio]);
   387→
   388→  // tiempo_agotado → siguiente desafío (o completado) tras 1.5s
   389→  useEffect(() => {
   390→    if (estado !== "tiempo_agotado") return;
   391→    const t = setTimeout(() => {
   392→      if (idxDesafio + 1 >= TOTAL_DESAFIOS) {
   393→        setEstado("completado");
   394→        silenciar();
   395→      } else {
   396→        setIdxDesafio((i) => i + 1);
   397→      }
   398→    }, DURACION_TIEMPO_AGOTADO_MS);
   399→    return () => clearTimeout(t);
   400→  }, [estado, idxDesafio]);
   401→
   402→  // fallando → vuelve a jugando tras 1.5s (no avanza)
   403→  useEffect(() => {
   404→    if (estado !== "fallando") return;
   405→    const t = setTimeout(() => {
   406→      setIdxBurbujaFallida(null);
   407→      setReglaVisible(false);
   408→      setEstado("jugando");
   409→    }, DURACION_FALLAR_MS);
   410→    return () => clearTimeout(t);
   411→  }, [estado]);
   412→
   413→  // Cuando idxDesafio cambia (excepto al iniciar en presentacion), cargar
   414→  // automáticamente el siguiente desafío.
   415→  // NOTA: NO incluir `estado` en las dependencias, porque eso dispararía
   416→  // `cargarDesafio` en cada transición de estado (acertando→jugando,
   417→  // fallando→jugando) y reiniciaría el desafío actual, impidiendo el avance.
   418→  // Solo nos interesa reaccionar al cambio de `idxDesafio`.
   419→  useEffect(() => {
   420→    if (estado === "presentacion" || estado === "completado") return;
   421→    cargarDesafio(idxDesafio);
   422→    // eslint-disable-next-line react-hooks/exhaustive-deps
   423→  }, [idxDesafio]);
   424→
   425→  /* ----------------------- Acciones ----------------------- */
   426→
   427→  const empezar = useCallback(() => {
   428→    setIdxDesafio(0);
   429→    setPuntos(0);
   430→    setAciertos(0);
   431→    setErrores(0);
   432→    setStreak(0);
   433→    setStreakMax(0);
   434→    cargarDesafio(0);
   435→  }, [cargarDesafio]);
   436→
   437→  const manejarToque = useCallback(
   438→    (b: Burbuja) => {
   439→      if (estado !== "jugando" && estado !== "fallando") return;
   440→      if (b.esCorrecta) {
   441→        // ¡Acierto!
   442→        setIdxBurbujaAtrapada(b.id);
   443→        setEstado("acertando");
   444→        setPuntos((p) => p + 10);
   445→        setAciertos((a) => a + 1);
   446→        setStreak((s) => {
   447→          const ns = s + 1;
   448→          setStreakMax((m) => Math.max(m, ns));
   449→          return ns;
   450→        });
   451→        setReglaVisible(false);
   452→        setIdxBurbujaFallida(null);
   453→      } else {
   454→        // Error: la burbuja tocada tiembla.
   455→        setIdxBurbujaFallida(b.id);
   456→        setEstado("fallando");
   457→        setPuntos((p) => Math.max(0, p - 3));
   458→        setErrores((e) => e + 1);
   459→        setStreak(0);
   460→        setReglaVisible(true);
   461→      }
   462→    },
   463→    [estado],
   464→  );
   465→
   466→  const reiniciar = useCallback(() => {
   467→    silenciar();
   468→    setEstado("presentacion");
   469→    setIdxDesafio(0);
   470→    setBurbujas([]);
   471→    setPuntos(0);
   472→    setAciertos(0);
   473→    setErrores(0);
   474→    setStreak(0);
   475→    setStreakMax(0);
   476→    setReglaVisible(false);
   477→    setIdxBurbujaAtrapada(null);
   478→    setIdxBurbujaFallida(null);
   479→    tiempoRestanteRef.current = TIEMPO_POR_PALABRA;
   480→    setTiempoRestante(TIEMPO_POR_PALABRA);
   481→  }, []);
   482→
   483→  const escuchar = useCallback(() => {
   484→    if (desafio) hablar(desafio.audio_guia);
   485→  }, [desafio]);
   486→
   487→  /* ----------------------- Derivados de render ----------------------- */
   488→
   489→  const porcentajeTiempo = (tiempoRestante / TIEMPO_POR_PALABRA) * 100;
   490→  const colorBarraTiempo =
   491→    porcentajeTiempo > 50
   492→      ? "from-emerald-400 to-teal-400"
   493→      : porcentajeTiempo > 25
   494→        ? "from-amber-400 to-orange-400"
   495→        : "from-rose-400 to-red-400";
   496→
   497→  const silabasPalabra = useMemo(() => {
   498→    if (!desafio) return [] as { texto: string; tonica: boolean }[];
   499→    return desafio.palabra_incompleta.split("-").map((s, i, arr) => {
   500→      // silaba_tonica: 1=última, 2=penúltima, 3=antepenúltima.
   501→      const tonicaIdxDesdeElFinal = desafio.silaba_tonica;
   502→      const idxTonica = arr.length - tonicaIdxDesdeElFinal;
   503→      return { texto: s, tonica: i === idxTonica };
   504→    });
   505→  }, [desafio]);
   506→
   507→  const silabasPalabraCompleta = useMemo(() => {
   508→    if (!desafio) return null;
   509→    // Mostrar la palabra completa con la tilde resaltada en verde.

   850→/* ----------------------- Aria helper ----------------------- */
   851→
   852→function ariaVocal(v: string): string {
   853→  // Describe la vocal: "é con tilde" o "a sin tilde".
   854→  const conTilde = "áéíóú".includes(v.toLowerCase());
   855→  const base = v.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
   856→  if (conTilde) return `Vocal ${v} con tilde`;
   857→  return `Vocal ${v} sin tilde (base ${base})`;
   858→}
