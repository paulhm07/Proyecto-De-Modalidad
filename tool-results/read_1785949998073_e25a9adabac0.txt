    50→/* ----------------------- Helpers ----------------------- */
    51→
    52→// Quita signos de puntuación de los extremos de un token.
    53→function limpiar(token: string): string {
    54→  return token
    55→    .replace(/^[.,;:!¡¿()"']+/, "")
    56→    .replace(/[.,;:!¡¿()"']+$/, "");
    57→}
    58→
    59→// Normaliza una frase para comparar (minúsculas, sin puntuación, sin dobles espacios).
    60→function normalizar(s: string): string {
    61→  return s
    62→    .toLowerCase()
    63→    .trim()
    64→    .split(/\s+/)
    65→    .map(limpiar)
    66→    .filter(Boolean)
    67→    .join(" ");
    68→}
    69→
    70→// ¿Es un nivel de pronombres? (opciones múltiples, no enunciado)
    71→function esNivelPronombres(tipo: NivelBus["tipo_ejercicio"]): boolean {
    72→  return tipo === "pronombre_el_ella" || tipo === "pronombre_ellos_ellas";
    73→}
    74→
    75→// Tokeniza un enunciado en palabras (limpiando puntuación de los extremos).
    76→function tokenizarEnunciado(enunciado: string): string[] {
    77→  return enunciado
    78→    .split(/\s+/)
    79→    .map((t) => limpiar(t))
    80→    .filter(Boolean);
    81→}
    82→
    83→// Fisher–Yates sin mutar el original.
    84→function mezclar<T>(arr: readonly T[]): T[] {
    85→  const a = [...arr];
    86→  for (let i = a.length - 1; i > 0; i--) {
    87→    const j = Math.floor(Math.random() * (i + 1));
    88→    [a[i], a[j]] = [a[j], a[i]];
    89→  }
    90→  return a;
    91→}

   258→/* ----------------------- Componente principal ----------------------- */
   259→
   260→export function BusLetras() {
   261→  const [nivelIdx, setNivelIdx] = useState(0);
   262→  const [estado, setEstado] = useState<Estado>("presentacion");
   263→  const [seleccionadas, setSeleccionadas] = useState<number[]>([]);
   264→  const [pasajerosTransportados, setPasajerosTransportados] = useState(0);
   265→  const [feedback, setFeedback] = useState<string | null>(null);
   266→
   267→  // Ref para timeouts internos (limpieza segura al desmontar)
   268→  const timeoutsRef = useRef<Set<ReturnType<typeof setTimeout>>>(new Set());
   269→  const programar = useCallback((fn: () => void, ms: number) => {
   270→    const t = setTimeout(() => {
   271→      timeoutsRef.current.delete(t);
   272→      fn();
   273→    }, ms);
   274→    timeoutsRef.current.add(t);
   275→  }, []);
   276→
   277→  useEffect(() => {
   278→    const set = timeoutsRef.current;
   279→    return () => {
   280→      set.forEach((t) => clearTimeout(t));
   281→      set.clear();
   282→    };
   283→  }, []);
   284→
   285→  const nivel = nivelesBus[nivelIdx];
   286→  const esPronombres = esNivelPronombres(nivel.tipo_ejercicio);
   287→
   288→  // Tokens del enunciado (memo por nivel)
   289→  const tokensEnunciado = useMemo(
   290→    () => tokenizarEnunciado(nivel.enunciado),
   291→    [nivel],
   292→  );
   293→
   294→  // Opciones de pronombres mezcladas (memo por nivel)
   295→  const opcionesPronombres = useMemo(
   296→    () =>
   297→      esPronombres
   298→        ? mezclar([nivel.palabra_correcta, ...nivel.distractores])
   299→        : [],
   300→    // eslint-disable-next-line react-hooks/exhaustive-deps
   301→    [nivel.nivel],
   302→  );
   303→
   304→  /* ---- Transición: celebrando → siguiente nivel o completado ---- */
   305→  useEffect(() => {
   306→    if (estado !== "celebrando") return;
   307→    const t = setTimeout(() => {
   308→      if (nivelIdx + 1 >= TOTAL_NIVELES) {
   309→        setEstado("completado");
   310→      } else {
   311→        setNivelIdx((n) => n + 1);
   312→        setSeleccionadas([]);
   313→        setFeedback(null);
   314→        setEstado("presentacion");
   315→      }
   316→    }, 2800);
   317→    return () => clearTimeout(t);
   318→  }, [estado, nivelIdx]);
   319→
   320→  /* ---- Validación ---- */
   321→  const validar = useCallback((): boolean => {
   322→    if (seleccionadas.length === 0) return false;
   323→    if (esPronombres) {
   324→      if (seleccionadas.length !== 1) return false;
   325→      return (
   326→        normalizar(opcionesPronombres[seleccionadas[0]]) ===
   327→        normalizar(nivel.palabra_correcta)
   328→      );
   329→    }
   330→    // Modo enunciado: ordenar índices asc y unir tokens
   331→    const frase = [...seleccionadas]
   332→      .sort((a, b) => a - b)
   333→      .map((i) => tokensEnunciado[i])
   334→      .join(" ");
   335→    return normalizar(frase) === normalizar(nivel.palabra_correcta);
   336→  }, [seleccionadas, esPronombres, opcionesPronombres, nivel.palabra_correcta, tokensEnunciado]);
   337→
   338→  /* ---- Acciones ---- */
   339→  const empezar = useCallback(() => {
   340→    setEstado("seleccionando");
   341→    setFeedback(null);
   342→  }, []);
   343→
   344→  const toggleSeleccion = useCallback(
   345→    (i: number) => {
   346→      if (estado !== "seleccionando") return;
   347→      setFeedback(null);
   348→      if (esPronombres) {
   349→        // En pronombres solo se permite 1 selección a la vez
   350→        setSeleccionadas((p) => (p.includes(i) ? [] : [i]));
   351→        return;
   352→      }
   353→      setSeleccionadas((p) =>
   354→        p.includes(i) ? p.filter((x) => x !== i) : [...p, i],
   355→      );
   356→    },
   357→    [estado, esPronombres],
   358→  );
   359→
   360→  const subirAlBus = useCallback(() => {
   361→    if (estado !== "seleccionando") return;
   362→    if (seleccionadas.length === 0) return;
   363→    setEstado("verificando");
   364→    const correcto = validar();
   365→    programar(() => {
   366→      if (correcto) {
   367→        setPasajerosTransportados((v) => v + 1);
   368→        setEstado("celebrando");
   369→      } else {
   370→        setFeedback(nivel.feedback_error);
   371→        setEstado("seleccionando");
   372→      }
   373→    }, 550);
   374→  }, [estado, seleccionadas.length, validar, nivel.feedback_error, programar]);
   375→
   376→  const limpiarSeleccion = useCallback(() => {
   377→    setSeleccionadas([]);
   378→    setFeedback(null);
   379→  }, []);
   380→
   381→  const reiniciarJuego = useCallback(() => {
   382→    setNivelIdx(0);
   383→    setEstado("presentacion");
   384→    setSeleccionadas([]);
   385→    setPasajerosTransportados(0);
   386→    setFeedback(null);
   387→  }, []);
   388→
   389→  /* ---- Derivados ---- */
   390→  const progresoNivel = useMemo(
   391→    () => ((nivelIdx + (estado === "celebrando" ? 1 : 0)) / TOTAL_NIVELES) * 100,
   392→    [nivelIdx, estado],
   393→  );
   394→  const esPresentacion = estado === "presentacion";
   395→  const esSeleccionando = estado === "seleccionando";
   396→  const esVerificando = estado === "verificando";
   397→  const esCelebrando = estado === "celebrando";
   398→  const puedeSubir = esSeleccionando && seleccionadas.length > 0;
   399→
   400→  /* ===================== RENDER ===================== */
   401→
   402→  return (
   403→    <div className="relative min-h-[calc(100vh-72px)] overflow-hidden bg-gradient-to-br from-rose-100 via-amber-100 to-rose-200 font-display">
   404→      {/* Cielo: sol y nubes */}
   405→      <div aria-hidden className="pointer-events-none absolute inset-0">
   406→        <div className="absolute right-6 top-8 h-16 w-16 rounded-full bg-amber-300 opacity-80 blur-sm sm:right-10 sm:top-12 sm:h-20 sm:w-20" />
   407→        <div className="absolute right-12 top-10 h-12 w-12 rounded-full bg-amber-200 opacity-70 sm:right-16 sm:top-16 sm:h-14 sm:w-14" />
   408→        <div className="absolute left-1/4 top-6 h-6 w-16 rounded-full bg-white/70 blur-[1px]" />
   409→        <div className="absolute right-1/4 top-14 h-5 w-14 rounded-full bg-white/60 blur-[1px]" />
   410→      </div>
   411→
   412→      {/* Palmeras a los lados (solo desktop) */}
   413→      <Palmera className="pointer-events-none absolute -left-2 bottom-2 z-0 hidden h-56 w-32 -rotate-6 opacity-90 sm:block lg:h-72 lg:w-40" />
   414→      <Palmera className="pointer-events-none absolute -right-2 bottom-2 z-0 hidden h-56 w-32 rotate-6 opacity-90 sm:block lg:h-72 lg:w-40" />
   415→
   416→      <div className="relative z-10 mx-auto max-w-5xl px-3 pb-6 pt-4 sm:px-5 sm:pt-6">
   417→        {/* ===================== HUD superior ===================== */}
   418→        <div className="mb-4 rounded-2xl border-2 border-rose-200 bg-white/70 p-3 shadow-md backdrop-blur sm:p-4">
   419→          <div className="flex items-center justify-between gap-3">
   420→            <div className="flex items-center gap-2">
   421→              <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-rose-500 to-fuchsia-500 text-white shadow sm:h-11 sm:w-11">
   422→                <BusIcon size={20} strokeWidth={2.5} />
   423→              </div>
   424→              <div>
   425→                <p className="text-xs font-bold uppercase tracking-wide text-rose-700/70">
   426→                  El Bus de las Letras
   427→                </p>
   428→                <p className="text-base font-bold text-rose-950 sm:text-lg">
   429→                  Nivel {nivel.nivel}{" "}
   430→                  <span className="text-rose-600/60">de {TOTAL_NIVELES}</span>
   431→                </p>
   432→              </div>
   433→            </div>
   434→            <div className="flex items-center gap-2">
   435→              <div className="flex items-center gap-1.5 rounded-full border-2 border-amber-300 bg-amber-50 px-3 py-1.5">
   436→                <Users size={15} className="text-amber-700" strokeWidth={2.5} />
   437→                <span className="text-sm font-bold text-amber-800">
   438→                  {pasajerosTransportados}{" "}
   439→                  <span className="hidden sm:inline">transportados</span>
   440→                </span>
   441→              </div>
   442→            </div>
   443→          </div>
   444→          {/* Barra de progreso */}
   445→          <div className="mt-2.5 h-2.5 w-full overflow-hidden rounded-full bg-rose-200/70">
   446→            <div
   447→              className="h-full rounded-full bg-gradient-to-r from-rose-400 via-fuchsia-400 to-amber-400 transition-all duration-700 ease-out"
   448→              style={{ width: `${progresoNivel}%` }}
   449→            />
   450→          </div>
   451→        </div>
   452→
   453→        {/* ===================== Pantalla: completado ===================== */}
   454→        {estado === "completado" ? (
   455→          <PantallaCompletado
   456→            onReiniciar={reiniciarJuego}
   457→            pasajeros={pasajerosTransportados}
   458→          />
   459→        ) : esPresentacion ? (
   460→          /* ===================== Pantalla: presentación ===================== */
   461→          <div className="animate-bounce-in rounded-3xl border-2 border-rose-300 bg-white/85 p-6 text-center shadow-lg backdrop-blur sm:p-8">
   462→            <p className="text-xs font-bold uppercase tracking-wider text-rose-600">
   463→              Parada {nivel.nivel} de {TOTAL_NIVELES}
   464→            </p>
   465→            <h2 className="mt-1 flex items-center justify-center gap-1.5 text-2xl font-bold text-rose-950 sm:text-3xl">
   466→              <MapPin size={22} className="text-rose-500" strokeWidth={2.5} />
   467→              {nivel.parada}
   468→            </h2>
   469→
   470→            <div className="mx-auto my-5 flex h-24 w-24 items-center justify-center rounded-full bg-gradient-to-br from-amber-100 to-rose-100 text-6xl shadow-inner sm:h-28 sm:w-28">
   471→              {nivel.avatar_pasajero}
   472→            </div>
   473→
   474→            <p className="text-base font-bold text-rose-900 sm:text-lg">
   475→              “¡Buenas! Vengo esperando el bus.”
   476→            </p>
   477→            <div className="mx-auto mt-3 max-w-md rounded-2xl border-2 border-rose-200 bg-rose-50 px-4 py-3">
   478→              <p className="text-sm font-semibold text-rose-700 sm:text-base">
   479→                <Sparkles size={14} className="mr-1 inline" strokeWidth={2.5} />
   480→                {nivel.consigna_para_nino}
   481→              </p>
   482→            </div>
   483→
   484→            <button
   485→              onClick={empezar}
   486→              className="mt-6 flex w-full items-center justify-center gap-2 rounded-2xl bg-gradient-to-r from-rose-500 to-fuchsia-500 py-3.5 text-base font-bold text-white shadow-md transition hover:scale-[1.02] hover:shadow-lg active:scale-95 sm:w-auto sm:px-10"
   487→            >
   488→              <Sparkles size={18} strokeWidth={2.5} /> ¡Recoger al pasajero!
   489→              <ArrowRight size={18} strokeWidth={2.5} />
   490→            </button>
   491→          </div>
   492→        ) : (
   493→          /* ===================== Pantalla: jugando ===================== */
   494→          <div className="grid gap-4 lg:grid-cols-2">
   495→            {/* ============ Columna izquierda: escena (pasajero + bus) ============ */}
   496→            <div className="flex flex-col gap-4">
   497→              {/* Pasajero con bocadillo */}
   498→              <div className="rounded-3xl border-2 border-rose-200 bg-white/85 p-4 shadow-lg backdrop-blur sm:p-5">
   499→                <div className="flex items-start gap-3">
   500→                  <div
   501→                    className={`flex h-16 w-16 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-100 to-rose-100 text-4xl shadow-inner transition-all duration-700 ease-out sm:h-20 sm:w-20 sm:text-5xl ${
   502→                      esCelebrando
   503→                        ? "translate-x-6 -translate-y-4 scale-50 opacity-0"
   504→                        : ""
   505→                    }`}
   506→                    aria-label="Pasajero esperando el bus"
   507→                  >
   508→                    {nivel.avatar_pasajero}
   509→                  </div>
   510→                  <div className="relative flex-1 rounded-2xl border-2 border-rose-300 bg-rose-50 px-4 py-3 shadow-sm">
   511→                    <div className="absolute -left-2 top-5 h-4 w-4 rotate-45 border-b-2 border-l-2 border-rose-300 bg-rose-50" />
   512→                    <p className="text-base font-semibold leading-snug text-rose-950 sm:text-lg">
   513→                      {nivel.consigna_para_nino}
   514→                    </p>
   515→                  </div>
   516→                </div>
   517→                <p className="mt-3 flex items-center justify-center gap-1 text-center text-xs font-semibold text-rose-700/80 sm:text-sm">
   518→                  <MapPin size={12} strokeWidth={2.5} /> Parada:{" "}
   519→                  <span className="font-bold text-rose-800">{nivel.parada}</span>
   520→                </p>
   521→              </div>
   522→
   523→              {/* Letrero de parada */}
   524→              <LetreroParada texto={nivel.parada} />
   525→
   526→              {/* Bus SVG */}
   527→              <div className="rounded-3xl border-2 border-amber-300 bg-white/70 p-3 shadow-lg backdrop-blur sm:p-4">
   528→                <BusSVG completo={esCelebrando} />
   529→                <p className="mt-1 text-center text-xs font-semibold text-rose-700/70">
   530→                  {esCelebrando
   531→                    ? "¡Pasajero a bordo! 🎉"
   532→                    : "Bus escolar esperando pasajeros…"}
   533→                </p>
   534→              </div>
   535→            </div>
   536→
   537→            {/* ============ Columna derecha: palabras + feedback + botones ============ */}
   538→            <div className="flex flex-col gap-4">
   539→              {/* Tarjetas de palabra (enunciado o pronombres) */}
   540→              <div className="rounded-3xl border-2 border-rose-300 bg-white/85 p-4 shadow-lg backdrop-blur sm:p-5">
   541→                <div className="mb-2 flex items-center justify-between">
   542→                  <h3 className="flex items-center gap-1.5 text-sm font-bold text-rose-950 sm:text-base">

   544→                    {esPronombres ? "¿Qué pronombre usamos?" : "Oración del pasajero"}
   545→                  </h3>
   546→                  <span className="rounded-full bg-rose-100 px-2.5 py-0.5 text-xs font-bold text-rose-800">

   548→                  </span>
   549→                </div>
   550→
   551→                {!esPronombres && (
   552→                  <p className="mb-3 text-center text-xs font-semibold text-rose-700/80 sm:text-sm">
   553→                    <Hand size={13} className="mr-1 inline" strokeWidth={2.5} />
   554→                    Tocá la palabra (o palabras) correcta y luego{" "}
   555→                    <b>“Subir al bus”</b>.
   556→                  </p>

   559→                  <p className="mb-3 text-center text-xs font-semibold text-rose-700/80 sm:text-sm">
   560→                    <Hand size={13} className="mr-1 inline" strokeWidth={2.5} />

   562→                    <b>“Subir al bus”</b>.
   563→                  </p>
   564→                )}
   565→
   566→                {/* Lista de palabras / opciones */}
   567→                {esPronombres ? (
   568→                  <div className="flex flex-wrap items-center justify-center gap-3">
   569→                    {opcionesPronombres.map((opcion, i) => {
   570→                      const sel = seleccionadas.includes(i);
   571→                      return (
   572→                        <button
   573→                          key={i}
   574→                          onClick={() => toggleSeleccion(i)}
   575→                          disabled={!esSeleccionando}
   576→                          className={`select-none rounded-xl border-2 px-6 py-3 text-lg font-bold transition-all sm:text-xl ${

   578→                              ? "scale-[1.05] border-rose-500 bg-rose-100 text-rose-900 shadow-md"
   579→                              : "border-rose-300 bg-white/90 text-rose-950 hover:border-rose-400 hover:bg-rose-50"
   580→                          } ${esSeleccionando ? "cursor-pointer active:scale-95" : "cursor-default"}`}
   581→                        >
   582→                          {opcion}
   583→                        </button>
   584→                      );
   585→                    })}
   586→                  </div>
   587→                ) : (
   588→                  <div className="flex flex-wrap items-center justify-center gap-2">
   589→                    {tokensEnunciado.map((palabra, i) => {
   590→                      const sel = seleccionadas.includes(i);
   591→                      return (
   592→                        <button
   593→                          key={i}
   594→                          onClick={() => toggleSeleccion(i)}
   595→                          disabled={!esSeleccionando}
   596→                          className={`select-none rounded-xl border-2 px-3.5 py-2.5 text-base font-bold transition-all sm:px-4 sm:text-lg ${
   597→                            sel
   598→                              ? "scale-[1.05] border-rose-500 bg-rose-100 text-rose-900 shadow-md"
   599→                              : "border-rose-300 bg-white/90 text-rose-950 hover:border-rose-400 hover:bg-rose-50"
   600→                          } ${esSeleccionando ? "cursor-pointer active:scale-95" : "cursor-default"}`}
   601→                        >
   602→                          {palabra}
   603→                        </button>
   604→                      );
   605→                    })}
   606→                  </div>
   607→                )}
   608→
   609→                {/* Frase construida (preview) */}
   610→                {seleccionadas.length > 0 && (
   611→                  <div className="mt-3 rounded-xl bg-rose-50/80 px-3 py-2 text-center">
   612→                    <p className="text-[11px] font-bold uppercase tracking-wide text-rose-600/70">
   613→                      Subiendo al bus:
   614→                    </p>
   615→                    <p className="text-base font-bold text-rose-900 sm:text-lg">
   616→                      {esPronombres
   617→                        ? opcionesPronombres[seleccionadas[0]]
   618→                        : [...seleccionadas]
   619→                            .sort((a, b) => a - b)
   620→                            .map((i) => tokensEnunciado[i])
   621→                            .join(" ")}
   622→                    </p>
   623→                  </div>
   624→                )}
   625→              </div>
   626→
   627→              {/* Feedback de error */}
   628→              {feedback && esSeleccionando && (
   629→                <div className="animate-bounce-in rounded-2xl border-2 border-amber-300 bg-amber-50 px-4 py-3 shadow">
   630→                  <p className="flex items-start gap-2 text-sm font-semibold text-amber-900">
   631→                    <span className="text-lg">💡</span>
   632→                    <span>{feedback}</span>
   633→                  </p>
   634→                </div>
   635→              )}
   636→
   637→              {/* Botones de acción */}
   638→              <div className="flex flex-col gap-2 sm:flex-row">
   639→                <button
   640→                  onClick={subirAlBus}
   641→                  disabled={!puedeSubir || esVerificando}
   642→                  className={`flex flex-1 items-center justify-center gap-2 rounded-2xl py-3.5 text-base font-bold shadow-md transition active:scale-95 ${
   643→                    puedeSubir
   644→                      ? "bg-gradient-to-r from-rose-500 to-fuchsia-500 text-white hover:scale-[1.02] hover:shadow-lg"
   645→                      : "cursor-not-allowed bg-rose-200 text-rose-500/70"
   646→                  }`}
   647→                >
   648→                  {esVerificando ? (
   649→                    <>
   650→                      <RefreshCw size={18} className="animate-spin" strokeWidth={2.5} />{" "}
   651→                      Revisando…
   652→                    </>
   653→                  ) : (
   654→                    <>
   655→                      <Check size={18} strokeWidth={2.5} /> ¡Subir al bus!
   656→                    </>
   657→                  )}
   658→                </button>
   659→
   660→                {seleccionadas.length > 0 && esSeleccionando && (
   661→                  <button
   662→                    onClick={limpiarSeleccion}
   663→                    className="flex items-center justify-center gap-1.5 rounded-2xl border-2 border-rose-300 bg-white py-3.5 px-4 text-sm font-bold text-rose-700 transition hover:bg-rose-50 active:scale-95 sm:text-base"
   664→                    aria-label="Limpiar la selección de palabras"
   665→                  >
   666→                    <RotateCcw size={16} strokeWidth={2.5} /> Limpiar
   667→                  </button>
   668→                )}
   669→
   670→                {esCelebrando && (
   671→                  <button
   672→                    disabled
   673→                    className="flex flex-1 items-center justify-center gap-2 rounded-2xl bg-gradient-to-r from-emerald-400 to-lime-500 py-3.5 text-base font-bold text-white shadow-md"
   674→                  >
   675→                    <PartyPopper size={18} strokeWidth={2.5} /> ¡Bien hecho!
   676→                  </button>
   677→                )}
   678→              </div>
   679→            </div>
   680→          </div>
   681→        )}
   682→
   683→        {/* Overlay de celebración */}
   684→        {esCelebrando && (
   685→          <div className="pointer-events-none absolute inset-0 z-20 flex items-start justify-center pt-24 sm:pt-32">
   686→            <div className="animate-bounce-in rounded-3xl border-2 border-emerald-300 bg-white/95 px-6 py-4 shadow-2xl">
   687→              <p className="text-center text-sm font-bold uppercase tracking-wide text-emerald-700">
   688→                ¡Pasajero a bordo!
   689→              </p>
   690→              <p className="mt-1 text-center text-2xl font-bold text-emerald-900 sm:text-3xl">
   691→                {nivel.palabra_correcta}
   692→              </p>
   693→              <p className="mt-1 text-center text-xs font-semibold text-emerald-700 sm:text-sm">
   694→                ¡Muy bien! El pasajero subió al bus.
   695→              </p>
   696→            </div>
   697→          </div>
   698→        )}
   699→      </div>
   700→
   701→      {/* Confeti al celebrar */}
   702→      {esCelebrando && <Confetti duracionMs={2800} cantidad={48} />}
   703→    </div>
   704→  );
   705→}
