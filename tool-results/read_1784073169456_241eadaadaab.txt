   631→function StatBox({
   632→  label,
   633→  value,
   634→  icon,
   635→  tint,
   636→}: {
   637→  label: string;
   638→  value: number;
   639→  icon: React.ReactNode;
   640→  tint: "cyan" | "amber" | "rose";
   641→}) {
   642→  const tintMap = {
   643→    cyan: { border: "border-cyan-400/40", bg: "bg-cyan-400/10", text: "text-cyan-200", glow: "text-cyan-300" },
   644→    amber: { border: "border-amber-400/40", bg: "bg-amber-400/10", text: "text-amber-200", glow: "text-amber-300" },
   645→    rose: { border: "border-rose-400/40", bg: "bg-rose-400/10", text: "text-rose-200", glow: "text-rose-300" },
   646→  }[tint];

   655→    </div>
   656→  );
