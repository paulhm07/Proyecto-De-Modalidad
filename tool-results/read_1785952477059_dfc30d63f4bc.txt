    97→    97→function hablar(texto: string) {
    98→    98→  if (typeof window === "undefined") return;
    99→    99→  if (!("speechSynthesis" in window)) return;
   100→   100→  try {
   101→   101→    window.speechSynthesis.cancel();
   102→   102→    const u = new SpeechSynthesisUtterance(normalizarAudioGuia(texto));
   103→   103→    u.lang = "es-ES";
   104→   104→    u.rate = 0.85;
   105→   105→    u.pitch = 1.1;
   106→   106→    window.speechSynthesis.speak(u);
   107→   107→  } catch {
   108→   108→    /* noop */
   109→   109→  }
   110→   110→}

   123→   123→
   124→   124→function slotAX(
   125→   125→  slot: "izquierda" | "centro" | "derecha",
   126→   126→  ancho: number,
   127→   127→  size: number,
   128→   128→): number {
   129→   129→  if (slot === "izquierda") return Math.max(8, ancho * 0.12);
   130→   130→  if (slot === "centro") return Math.max(8, ancho * 0.5 - size / 2);
   131→   131→  return Math.max(8, ancho * 0.88 - size);
   132→   132→}
   133→   133→
   134→   134→function slotBY(slotIdx: number, alto: number, size: number): number {
   135→   135→  // Tres filas virtuales para espaciar las burbujas al iniciar.
   136→   136→  const filas = [alto * 0.18, alto * 0.45, alto * 0.72];
   137→   137→  const f = filas[slotIdx % filas.length];
   138→   138→  return Math.min(Math.max(8, f), Math.max(8, alto - size - 8));
   139→   139→}

   141→   141→function elegir<T>(arr: T[], n: number): T[] {
   142→   142→  const copia = [...arr];
   143→   143→  for (let i = copia.length - 1; i > 0; i--) {
   144→   144→    const j = Math.floor(Math.random() * (i + 1));
   145→   145→    [copia[i], copia[j]] = [copia[j], copia[i]];
   146→   146→  }
   147→   147→  return copia.slice(0, n);
   148→   148→}
   149→   149→
   150→   150→function shuffleThree<T>(arr: [T, T, T]): [T, T, T] {
   151→   151→  const copia = [...arr];
   152→   152→  for (let i = copia.length - 1; i > 0; i--) {
   153→   153→    const j = Math.floor(Math.random() * (i + 1));
   154→   154→    [copia[i], copia[j]] = [copia[j], copia[i]];
   155→   155→  }
   156→   156→  return [copia[0], copia[1], copia[2]];
   157→   157→}
   158→   158→
   159→   159→function shuffleTwo<T>(arr: [T, T]): [T, T] {
   160→   160→  return Math.random() < 0.5 ? [arr[0], arr[1]] : [arr[1], arr[0]];
   161→   161→}

   180→   180→
   181→   181→  // Ref espejo de tiempoRestante para poder leer el valor actual dentro
   182→   182→  // del callback del interval sin reiniciar el effect.

   184→   184→
   185→   185→  const containerRef = useRef<HTMLDivElement>(null);

   187→   187→  const [esDesktop, setEsDesktop] = useState(false);
   188→   188→  const bubbleSize = esDesktop ? 96 : 80;
   189→   189→
   190→   190→  // Set de timeouts para limpieza segura al desmontar.

   192→   192→  const programar = useCallback((fn: () => void, ms: number) => {
   193→   193→    const t = setTimeout(() => {
   194→   194→      timeoutsRef.current.delete(t);
   195→   195→      fn();

   223→   223→  }, []);
   224→   224→

   226→   226→  useEffect(() => {

   251→   251→        ["izquierda", "centro", "derecha"] as const
   252→   252→      ).filter((s) => s !== d.posicion_x);
   253→   253→      const [s1, s2] = shuffleTwo([slotsRestantes[0], slotsRestantes[1]]);
   254→   254→      const vocalesSlots: { vocal: string; esCorrecta: boolean; slot: "izquierda" | "centro" | "derecha" }[] = [
   255→   255→        { vocal: d.letra_con_tilde_correcta, esCorrecta: true, slot: d.posicion_x },
   256→   256→        { vocal: d.distractores[0], esCorrecta: false, slot: s1 },

   271→   271→          -Math.PI / 4 + (Math.random() - 0.5) * 0.6, // ↗

   275→   275→          id: i,
   276→   276→          vocal: v.vocal,
   277→   277→          esCorrecta: v.esCorrecta,
   278→   278→          x: slotAX(v.slot, w, size),
   279→   279→          y: slotBY(i, h, size),
   280→   280→          vx: Math.cos(ang) * speed,
   281→   281→          vy: Math.sin(ang) * speed,

   303→   303→  useEffect(() => {

   305→   305→      estado !== "jugando" &&
   306→   306→      estado !== "fallando" &&
   307→   307→      estado !== "acertando"

   310→   310→    }

   315→   315→      last = now;
   316→   316→      const { w, h } = dimsRef.current;
   317→   317→      const size = window.matchMedia("(min-width: 640px)").matches ? 96 : 80;
   318→   318→      const maxX = Math.max(0, w - size);
   319→   319→      const maxY = Math.max(0, h - size);

   321→   321→        prev.map((b) => {
   322→   322→          if (estado === "acertando" && b.id === idxBurbujaAtrapada) {
   323→   323→            // La burbuja atrapada se queda quieta (luego "explota").
   324→   324→            return b;
   325→   325→          }

   327→   327→          x += vx * dt;
   328→   328→          y += vy * dt;
   329→   329→          if (x < 0) {
   330→   330→            x = 0;
   331→   331→            vx = -vx;
   332→   332→          } else if (x > maxX) {
   333→   333→            x = maxX;
   334→   334→            vx = -vx;
   335→   335→          }
   336→   336→          if (y < 0) {
   337→   337→            y = 0;
   338→   338→            vy = -vy;
   339→   339→          } else if (y > maxY) {
   340→   340→            y = maxY;
   341→   341→            vy = -vy;
   342→   342→          }
   343→   343→          return { ...b, x, y, vx, vy };
   344→   344→        }),
   345→   345→      );
   346→   346→      raf = requestAnimationFrame(loop);
   347→   347→    };
   348→   348→    raf = requestAnimationFrame(loop);
   349→   349→    return () => cancelAnimationFrame(raf);
   350→   350→  }, [estado, idxBurbujaAtrapada]);
   351→   351→
   352→   352→  /* ----------------------- Timer (5s por palabra) ----------------------- */
   353→   353→  // Solo corre en 'jugando'. Se pausa durante 'fallando' y 'acertando'.
   354→   354→  // La transición a 'tiempo_agotado' se dispara DENTRO del callback del
   355→   355→  // interval (no en un effect body) para evitar renders en cascada.
   356→   356→
   357→   357→  useEffect(() => {
   358→   358→    if (estado !== "jugando") return;

   360→   360→      const next = Math.max(0, tiempoRestanteRef.current - 0.1);
   361→   361→      tiempoRestanteRef.current = next;
   362→   362→      setTiempoRestante(next);

   364→   364→        // Tiempo agotado: transición directa desde el callback.
   365→   365→        setStreak(0);
   366→   366→        setEstado("tiempo_agotado");
   367→   367→      }
   368→   368→    }, 100);
   369→   369→    return () => clearInterval(id);
   370→   370→  }, [estado]);
   371→   371→
   372→   372→  /* ----------------------- Transiciones de estado ----------------------- */
   373→   373→
   374→   374→  // acertando → siguiente desafío (o completado) tras 1.2s
   375→   375→  useEffect(() => {
   376→   376→    if (estado !== "acertando") return;

   378→   378→      if (idxDesafio + 1 >= TOTAL_DESAFIOS) {
   379→   379→        setEstado("completado");
   380→   380→        silenciar();
   381→   381→      } else {
   382→   382→        setIdxDesafio((i) => i + 1);
   383→   383→      }
   384→   384→    }, DURACION_ACERTAR_MS);
   385→   385→    return () => clearTimeout(t);
   386→   386→  }, [estado, idxDesafio]);
   387→   387→
   388→   388→  // tiempo_agotado → siguiente desafío (o completado) tras 1.5s
   389→   389→  useEffect(() => {
   390→   390→    if (estado !== "tiempo_agotado") return;
   391→   391→    const t = setTimeout(() => {
   392→   392→      if (idxDesafio + 1 >= TOTAL_DESAFIOS) {
   393→   393→        setEstado("completado");
   394→   394→        silenciar();
   395→   395→      } else {
   396→   396→        setIdxDesafio((i) => i + 1);
   397→   397→      }
   398→   398→    }, DURACION_TIEMPO_AGOTADO_MS);
   399→   399→    return () => clearTimeout(t);
   400→   400→  }, [estado, idxDesafio]);
   401→   401→
   402→   402→  // fallando → vuelve a jugando tras 1.5s (no avanza)
   403→   403→  useEffect(() => {
   404→   404→    if (estado !== "fallando") return;
   405→   405→    const t = setTimeout(() => {
   406→   406→      setIdxBurbujaFallida(null);
   407→   407→      setReglaVisible(false);
   408→   408→      setEstado("jugando");
   409→   409→    }, DURACION_FALLAR_MS);
   410→   410→    return () => clearTimeout(t);
   411→   411→  }, [estado]);
   412→   412→
   413→   413→  // Cuando idxDesafio cambia (excepto al iniciar en presentacion), cargar
   414→   414→  // automáticamente el siguiente desafío.
   415→   415→  // NOTA: NO incluir `estado` en las dependencias, porque eso dispararía
   416→   416→  // `cargarDesafio` en cada transición de estado (acertando→jugando,
   417→   417→  // fallando→jugando) y reiniciaría el desafío actual, impidiendo el avance.
   418→   418→  // Solo nos interesa reaccionar al cambio de `idxDesafio`.
   419→   419→  useEffect(() => {
   420→   420→    if (estado === "presentacion" || estado === "completado") return;
   421→   421→    cargarDesafio(idxDesafio);
   422→   422→    // eslint-disable-next-line react-hooks/exhaustive-deps
   423→   423→  }, [idxDesafio]);
   424→   424→
   425→   425→  /* ----------------------- Acciones ----------------------- */
   426→   426→
   427→   427→  const empezar = useCallback(() => {
   428→   428→    setIdxDesafio(0);

   852→   852→function ariaVocal(v: string): string {
   853→   853→  // Describe la vocal: "é con tilde" o "a sin tilde".
   854→   854→  const conTilde = "áéíóú".includes(v.toLowerCase());
   855→   855→  const base = v.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
   856→   856→  if (conTilde) return `Vocal ${v} con tilde`;
   857→   857→  return `Vocal ${v} sin tilde (base ${base})`;
   858→   858→}
