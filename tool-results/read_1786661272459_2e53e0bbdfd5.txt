    35→function formatearFechaCorta(fecha: string): string {
    36→  try {
    37→    return new Date(fecha).toLocaleDateString("es-NI", {
    38→      day: "numeric",
    39→      month: "short",
    40→      year: "numeric",
    41→    });
    42→  } catch {
    43→    return fecha;
    44→  }
    45→}
    46→
    47→function colorNota(nota: number): { bg: string; texto: string } {
    48→  if (nota >= 70) return { bg: "bg-emerald-100", texto: "text-emerald-700" };
    49→  if (nota >= 50) return { bg: "bg-amber-100", texto: "text-amber-700" };
    50→  return { bg: "bg-rose-100", texto: "text-rose-700" };
    51→}

    53→function configEstado(estado: EstadoAvance): {

    57→  icon: typeof Check;
    58→  gradiente: string;
    59→} {
    60→  switch (estado) {
    61→    case "AL_DIA":
    62→      return {
    63→        label: "Al día",

    65→        texto: "text-emerald-700",

    67→        gradiente: "from-emerald-400 to-teal-500",

   106→function MedallaIcono({ iconoUrl, size = 28 }: { iconoUrl?: string | null; size?: number }) {
   107→  const esUrl = iconoUrl && (iconoUrl.startsWith("http") || iconoUrl.startsWith("/"));
   108→  const esEmoji = iconoUrl && !esUrl && iconoUrl.length <= 4;
   109→  if (esUrl) {
   110→    return (
   111→      <img
   112→        src={iconoUrl!}
   113→        alt=""
   114→        width={size}
   115→        height={size}
   116→        className="h-auto w-auto"
   117→        style={{ maxWidth: size, maxHeight: size }}
   118→        onError={(e) => {
   119→          (e.currentTarget as HTMLImageElement).style.display = "none";
   120→        }}
   121→      />
   122→    );
   123→  }
   124→  if (esEmoji) {
   125→    return <span style={{ fontSize: size }}>{iconoUrl}</span>;
   126→  }
   127→  return <Award size={size} className="text-amber-500" strokeWidth={2.5} />;
   128→}
