     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import { api } from "@/lib/api";
     5→import { useApp } from "@/context/AppContext";
     6→import type {
     7→  Asignatura,
     8→  Desafio,
     9→  Modulo,
    10→} from "@/lib/types";
    11→
    12→type Nivel = "asignaturas" | "modulos" | "desafios";
    13→
    14→interface OpcionDraft {
    15→  texto: string;
    16→  esCorrecta: boolean;
    17→}
    18→
    19→// Cada asignatura detectada por nombre tiene su color vibrante
    20→const ASIGNATURA_GRADIENTS: Record<string, string> = {
    21→  Matemáticas: "from-cyan-400 via-sky-500 to-blue-600",
    22→  "Lengua y Literatura": "from-pink-400 via-rose-500 to-red-500",
    23→  "Ciencias Naturales": "from-emerald-400 via-green-500 to-teal-600",
    24→  "Estudios Sociales": "from-amber-400 via-orange-500 to-rose-500",
    25→};
    26→const DEFAULT_GRADIENT = "from-violet-400 via-purple-500 to-fuchsia-600";
    27→
    28→function gradientPara(nombre: string) {
    29→  return ASIGNATURA_GRADIENTS[nombre] ?? DEFAULT_GRADIENT;
    30→}
    31→
    32→export function ContentManager() {
    33→  const { usuario, mostrarToast } = useApp();
    34→  const [nivel, setNivel] = useState<Nivel>("asignaturas");
    35→
    36→  const [asignaturas, setAsignaturas] = useState<Asignatura[]>([]);
    37→  const [asignaturaSel, setAsignaturaSel] = useState<Asignatura | null>(null);
    38→  const [modulos, setModulos] = useState<Modulo[]>([]);
    39→  const [moduloSel, setModuloSel] = useState<Modulo | null>(null);
    40→  const [desafios, setDesafios] = useState<Desafio[]>([]);
    41→
    42→  const [cargandoAsignaturas, setCargandoAsignaturas] = useState(true);
    43→  const [cargandoModulos, setCargandoModulos] = useState(false);
    44→  const [cargandoDesafios, setCargandoDesafios] = useState(false);
    45→
    46→  const [modalAsignatura, setModalAsignatura] = useState(false);
    47→  const [modalModulo, setModalModulo] = useState(false);
    48→  const [modalDesafio, setModalDesafio] = useState(false);
    49→
    50→  const cargarAsignaturas = useCallback(async () => {
    51→    setCargandoAsignaturas(true);
    52→    try {
    53→      const data = await api.obtenerAsignaturas();
    54→      setAsignaturas(data);
    55→    } catch (err) {
    56→      console.error(err);
    57→    } finally {
    58→      setCargandoAsignaturas(false);
    59→    }
    60→  }, []);
    61→
    62→  useEffect(() => {
    63→    cargarAsignaturas();
    64→  }, [cargarAsignaturas]);
    65→
    66→  const cargarModulos = useCallback(async (asigId: string) => {
    67→    setCargandoModulos(true);
    68→    try {
    69→      const data = await api.obtenerModulos(asigId);
    70→      setModulos(data);
    71→    } catch (err) {
    72→      console.error(err);
    73→    } finally {
    74→      setCargandoModulos(false);
    75→    }
    76→  }, []);
    77→
    78→  const cargarDesafios = useCallback(async (modId: string) => {
    79→    setCargandoDesafios(true);
    80→    try {
    81→      const data = await api.obtenerDesafios(modId);
    82→      setDesafios(data);
    83→    } catch (err) {
    84→      console.error(err);
    85→    } finally {
    86→      setCargandoDesafios(false);
    87→    }
    88→  }, []);
    89→
    90→  // === Acciones CRUD ===
    91→  const crearAsignatura = async (nombre: string, descripcion: string) => {
    92→    if (!usuario) return;
    93→    try {
    94→      await api.crearAsignatura(
    95→        nombre.trim(),
    96→        descripcion.trim() || null,
    97→        usuario.id,
    98→      );
    99→      mostrarToast("¡Asignatura creada! 🎉", "exito");
   100→      setModalAsignatura(false);
   101→      cargarAsignaturas();
   102→    } catch (err) {
   103→      mostrarToast(err instanceof Error ? err.message : "Error", "error");
   104→    }
   105→  };
   106→
   107→  const eliminarAsignatura = async (a: Asignatura) => {
   108→    if (
   109→      !confirm(
   110→        `¿Eliminar "${a.nombre}" y todos sus módulos y desafíos? Esta acción no se puede deshacer.`,
   111→      )
   112→    )
   113→      return;
   114→    try {
   115→      await api.eliminarAsignatura(a.id);
   116→      mostrarToast("Asignatura eliminada", "info");
   117→      if (asignaturaSel?.id === a.id) {
   118→        setAsignaturaSel(null);
   119→        setModulos([]);
   120→      }
   121→      cargarAsignaturas();
   122→    } catch (err) {
   123→      mostrarToast(err instanceof Error ? err.message : "Error", "error");
   124→    }
   125→  };
   126→
   127→  const crearModulo = async (titulo: string) => {
   128→    if (!asignaturaSel) return;
   129→    try {
   130→      await api.crearModulo(asignaturaSel.id, titulo.trim());
   131→      mostrarToast("¡Módulo creado! 🎉", "exito");
   132→      setModalModulo(false);
   133→      cargarModulos(asignaturaSel.id);
   134→    } catch (err) {
   135→      mostrarToast(err instanceof Error ? err.message : "Error", "error");
   136→    }
   137→  };
   138→
   139→  const eliminarModulo = async (m: Modulo) => {
   140→    if (!confirm(`¿Eliminar el módulo "${m.titulo}"?`)) return;
   141→    try {
   142→      await api.eliminarModulo(m.id);
   143→      mostrarToast("Módulo eliminado", "info");
   144→      if (moduloSel?.id === m.id) {
   145→        setModuloSel(null);
   146→        setDesafios([]);
   147→      }
   148→      if (asignaturaSel) cargarModulos(asignaturaSel.id);
   149→    } catch (err) {
   150→      mostrarToast(err instanceof Error ? err.message : "Error", "error");
   151→    }
   152→  };
   153→
   154→  const crearDesafio = async (data: {
   155→    tipo: string;
   156→    pregunta: string;
   157→    puntos: number;
   158→    opciones: OpcionDraft[];
   159→  }) => {
   160→    if (!moduloSel) return;
   161→    if (data.opciones.filter((o) => o.esCorrecta).length !== 1) {
   162→      mostrarToast("Debe haber exactamente una opción correcta 🎯", "error");
   163→      return;
   164→    }
   165→    try {
   166→      await api.crearDesafio({
   167→        moduloId: moduloSel.id,
   168→        tipo: data.tipo,
   169→        pregunta: data.pregunta.trim(),
   170→        puntos: data.puntos,
   171→        opciones: data.opciones
   172→          .filter((o) => o.texto.trim())
   173→          .map((o) => ({ texto: o.texto.trim(), esCorrecta: o.esCorrecta })),
   174→      });
   175→      mostrarToast("¡Desafío creado! 🎯", "exito");
   176→      setModalDesafio(false);
   177→      cargarDesafios(moduloSel.id);
   178→    } catch (err) {
   179→      mostrarToast(err instanceof Error ? err.message : "Error", "error");
   180→    }
   181→  };
   182→
   183→  const eliminarDesafio = async (d: Desafio) => {
   184→    if (!confirm(`¿Eliminar el desafío "${d.pregunta.slice(0, 40)}..."?`))
   185→      return;
   186→    try {
   187→      await api.eliminarDesafio(d.id);
   188→      mostrarToast("Desafío eliminado", "info");
   189→      if (moduloSel) cargarDesafios(moduloSel.id);
   190→    } catch (err) {
   191→      mostrarToast(err instanceof Error ? err.message : "Error", "error");
   192→    }
   193→  };
   194→
   195→  // === Navegación ===
   196→  const entrarAsignatura = (a: Asignatura) => {
   197→    setAsignaturaSel(a);
   198→    setModuloSel(null);
   199→    setDesafios([]);
   200→    cargarModulos(a.id);
   201→    setNivel("modulos");
   202→  };
   203→  const entrarModulo = (m: Modulo) => {
   204→    setModuloSel(m);
   205→    cargarDesafios(m.id);
   206→    setNivel("desafios");
   207→  };
   208→  const volverAAsignaturas = () => {
   209→    setAsignaturaSel(null);
   210→    setModulos([]);
   211→    setModuloSel(null);
   212→    setDesafios([]);
   213→    setNivel("asignaturas");
   214→  };
   215→  const volverAModulos = () => {
   216→    setModuloSel(null);
   217→    setDesafios([]);
   218→    setNivel("modulos");
   219→  };
   220→
   221→  const accentGrad = asignaturaSel
   222→    ? gradientPara(asignaturaSel.nombre)
   223→    : DEFAULT_GRADIENT;
   224→
   225→  return (
   226→    <div className="mx-auto max-w-4xl px-4 py-8">
   227→      {/* Encabezado + breadcrumb */}
   228→      <div className="mb-6">
   229→        <div className="mb-4 flex items-center gap-3">
   230→          <div className="animate-float text-5xl">🎨</div>
   231→          <div>
   232→            <h1 className="text-3xl font-black text-purple-900 sm:text-4xl">
   233→              ¡Taller del maestro!
   234→            </h1>
   235→            <p className="font-bold text-purple-600">
   236→              ✨ Crea aventuras, módulos y desafíos
   237→            </p>
   238→          </div>
   239→        </div>
   240→
   241→        <div className="flex flex-wrap items-center gap-1 text-sm font-black text-purple-400">
   242→          <button
   243→            onClick={volverAAsignaturas}
   244→            className={`rounded-lg px-2 py-1 transition-colors ${
   245→              nivel === "asignaturas"
   246→                ? "bg-purple-100 text-purple-700"
   247→                : "hover:bg-purple-50 hover:text-purple-600"
   248→            }`}
   249→          >
   250→            📚 Asignaturas
   251→          </button>
   252→          {asignaturaSel && (
   253→            <>
   254→              <span>→</span>
   255→              <button
   256→                onClick={volverAModulos}
   257→                className={`rounded-lg px-2 py-1 transition-colors ${
   258→                  nivel === "modulos"
   259→                    ? "bg-purple-100 text-purple-700"
   260→                    : "hover:bg-purple-50 hover:text-purple-600"
   261→                }`}
   262→              >
   263→                {asignaturaSel.nombre}
   264→              </button>
   265→            </>
   266→          )}
   267→          {moduloSel && (
   268→            <>
   269→              <span>→</span>
   270→              <span className="rounded-lg bg-purple-100 px-2 py-1 text-purple-700">
   271→                {moduloSel.titulo}
   272→              </span>
   273→            </>
   274→          )}
   275→        </div>
   276→      </div>
   277→
   278→      {/* NIVEL 1: ASIGNATURAS */}
   279→      {nivel === "asignaturas" && (
   280→        <>
   281→          <div className="mb-5 flex justify-end">
   282→            <button
   283→              onClick={() => setModalAsignatura(true)}
   284→              className="btn-3d rounded-2xl bg-gradient-to-r from-fuchsia-500 via-violet-500 to-purple-600 px-5 py-3 text-sm font-black text-white shadow-xl shadow-violet-200 transition-all hover:scale-105"
   285→            >
   286→              + Nueva asignatura 📚
   287→            </button>
   288→          </div>
   289→
   290→          {cargandoAsignaturas ? (
   291→            <div className="space-y-3">
   292→              {[0, 1, 2].map((i) => (
   293→                <div
   294→                  key={i}
   295→                  className="h-24 animate-pulse rounded-[1.5rem] bg-white/50"
   296→                />
   297→              ))}
   298→            </div>
   299→          ) : asignaturas.length === 0 ? (
   300→            <EmptyState
   301→              emoji="📭"
   302→              titulo="¡No hay asignaturas!"
   303→              subtitulo="Crea tu primera asignatura para empezar la aventura"
   304→              cta={{
   305→                label: "+ Crear asignatura",
   306→                onClick: () => setModalAsignatura(true),
   307→              }}
   308→            />
   309→          ) : (
   310→            <div className="space-y-3">
   311→              {asignaturas.map((a, i) => {
   312→                const grad = gradientPara(a.nombre);
   313→                return (
   314→                  <div
   315→                    key={a.id}
   316→                    style={{ animationDelay: `${i * 60}ms` }}
   317→                    className="animate-pop flex items-center gap-4 rounded-[1.5rem] border-4 border-white bg-white/80 p-4 shadow-lg backdrop-blur-sm"
   318→                  >
   319→                    <button
   320→                      onClick={() => entrarAsignatura(a)}
   321→                      className="group flex flex-1 items-center gap-4 text-left"
   322→                    >
   323→                      <div
   324→                        className={`flex h-14 w-14 items-center justify-center rounded-3xl bg-gradient-to-br ${grad} text-2xl shadow-lg transition-transform group-hover:scale-110`}
   325→                      >
   326→                        📚
   327→                      </div>
   328→                      <div className="flex-1">
   329→                        <p className="text-lg font-black text-purple-900">
   330→                          {a.nombre}
   331→                        </p>
   332→                        <p className="text-xs font-bold text-purple-500">
   333→                          {a.descripcion || "Sin descripción"} · 📦{" "}
   334→                          {a._count?.modulos ?? 0} módulo(s)
   335→                        </p>
   336→                      </div>
   337→                      <span className="text-2xl text-purple-300 transition-transform group-hover:translate-x-1">
   338→                        →
   339→                      </span>
   340→                    </button>
   341→                    <button
   342→                      onClick={() => eliminarAsignatura(a)}
   343→                      title="Eliminar asignatura"
   344→                      className="rounded-xl bg-red-100 p-2 text-red-400 transition-colors hover:bg-red-500 hover:text-white"
   345→                    >
   346→                      🗑️
   347→                    </button>
   348→                  </div>
   349→                );
   350→              })}
   351→            </div>
   352→          )}
   353→        </>
   354→      )}
   355→
   356→      {/* NIVEL 2: MÓDULOS */}
   357→      {nivel === "modulos" && asignaturaSel && (
   358→        <>
   359→          <div
   360→            className={`mb-5 rounded-[1.5rem] border-4 border-white bg-gradient-to-r ${accentGrad} p-4 shadow-lg`}
   361→          >
   362→            <h2 className="text-xl font-black text-white drop-shadow">
   363→              {asignaturaSel.nombre}
   364→            </h2>
   365→            <p className="text-sm font-bold text-white/90">
   366→              {asignaturaSel.descripcion || "Sin descripción"}
   367→            </p>
   368→          </div>
   369→
   370→          <div className="mb-5 flex justify-end">
   371→            <button
   372→              onClick={() => setModalModulo(true)}
   373→              className="btn-3d rounded-2xl bg-gradient-to-r from-fuchsia-500 via-violet-500 to-purple-600 px-5 py-3 text-sm font-black text-white shadow-xl shadow-violet-200 transition-all hover:scale-105"
   374→            >
   375→              + Nuevo módulo 📦
   376→            </button>
   377→          </div>
   378→
   379→          {cargandoModulos ? (
   380→            <div className="space-y-3">
   381→              {[0, 1].map((i) => (
   382→                <div
   383→                  key={i}
   384→                  className="h-24 animate-pulse rounded-[1.5rem] bg-white/50"
   385→                />
   386→              ))}
   387→            </div>
   388→          ) : modulos.length === 0 ? (
   389→            <EmptyState
   390→              emoji="📦"
   391→              titulo="¡Sin módulos!"
   392→              subtitulo="Crea el primer módulo de esta asignatura"
   393→              cta={{
   394→                label: "+ Crear módulo",
   395→                onClick: () => setModalModulo(true),
   396→              }}
   397→            />
   398→          ) : (
   399→            <div className="space-y-3">
   400→              {modulos.map((m, i) => (
   401→                <div
   402→                  key={m.id}
   403→                  style={{ animationDelay: `${i * 60}ms` }}
   404→                  className="animate-pop flex items-center gap-4 rounded-[1.5rem] border-4 border-white bg-white/80 p-4 shadow-lg backdrop-blur-sm"
   405→                >
   406→                  <button
   407→                    onClick={() => entrarModulo(m)}
   408→                    className="group flex flex-1 items-center gap-4 text-left"
   409→                  >
   410→                    <div
   411→                      className={`flex h-14 w-14 items-center justify-center rounded-3xl bg-gradient-to-br ${accentGrad} text-xl font-black text-white shadow-lg transition-transform group-hover:scale-110`}
   412→                    >
   413→                      {m.orden}
   414→                    </div>
   415→                    <div className="flex-1">
   416→                      <p className="text-lg font-black text-purple-900">
   417→                        {m.titulo}
   418→                      </p>
   419→                      <p className="text-xs font-bold text-purple-500">
   420→                        🧩 {m._count?.desafios ?? 0} desafío(s) · 🎯 Nivel
   421→                        mínimo {m.nivelMinimo}
   422→                      </p>
   423→                    </div>
   424→                    <span className="text-2xl text-purple-300 transition-transform group-hover:translate-x-1">
   425→                      →
   426→                    </span>
   427→                  </button>
   428→                  <button
   429→                    onClick={() => eliminarModulo(m)}
   430→                    title="Eliminar módulo"
   431→                    className="rounded-xl bg-red-100 p-2 text-red-400 transition-colors hover:bg-red-500 hover:text-white"
   432→                  >
   433→                    🗑️
   434→                  </button>
   435→                </div>
   436→              ))}
   437→            </div>
   438→          )}
   439→        </>
   440→      )}
   441→
   442→      {/* NIVEL 3: DESAFÍOS */}
   443→      {nivel === "desafios" && moduloSel && (
   444→        <>
   445→          <div
   446→            className={`mb-5 rounded-[1.5rem] border-4 border-white bg-gradient-to-r ${accentGrad} p-4 shadow-lg`}
   447→          >
   448→            <h2 className="text-xl font-black text-white drop-shadow">
   449→              {moduloSel.titulo}
   450→            </h2>
   451→            <p className="text-sm font-bold text-white/90">
   452→              🧩 {desafios.length} desafío(s) en este módulo
   453→            </p>
   454→          </div>
   455→
   456→          <div className="mb-5 flex justify-end">
   457→            <button
   458→              onClick={() => setModalDesafio(true)}
   459→              className="btn-3d rounded-2xl bg-gradient-to-r from-fuchsia-500 via-violet-500 to-purple-600 px-5 py-3 text-sm font-black text-white shadow-xl shadow-violet-200 transition-all hover:scale-105"
   460→            >
   461→              + Nuevo desafío 🎯
   462→            </button>
   463→          </div>
   464→
   465→          {cargandoDesafios ? (
   466→            <div className="space-y-3">
   467→              {[0, 1].map((i) => (
   468→                <div
   469→                  key={i}
   470→                  className="h-28 animate-pulse rounded-[1.5rem] bg-white/50"
   471→                />
   472→              ))}
   473→            </div>
   474→          ) : desafios.length === 0 ? (
   475→            <EmptyState
   476→              emoji="🎯"
   477→              titulo="¡Sin desafíos!"
   478→              subtitulo="Crea el primer desafío para este módulo"
   479→              cta={{
   480→                label: "+ Crear desafío",
   481→                onClick: () => setModalDesafio(true),
   482→              }}
   483→            />
   484→          ) : (
   485→            <div className="space-y-3">
   486→              {desafios.map((d, i) => (
   487→                <div
   488→                  key={d.id}
   489→                  style={{ animationDelay: `${i * 60}ms` }}
   490→                  className="animate-pop rounded-[1.5rem] border-4 border-white bg-white/80 p-4 shadow-lg backdrop-blur-sm"
   491→                >
   492→                  <div className="flex items-start gap-3">
   493→                    <span
   494→                      className={`mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br ${accentGrad} text-xs font-black text-white shadow`}
   495→                    >
   496→                      {i + 1}
   497→                    </span>
   498→                    <div className="flex-1">
   499→                      <p className="font-black text-purple-900">{d.pregunta}</p>
   500→                      <div className="mt-1 flex flex-wrap items-center gap-2 text-xs">
   501→                        <span className="rounded-full bg-amber-100 px-2 py-0.5 font-black text-amber-700">
   502→                          ⭐ {d.puntos} pts
   503→                        </span>
   504→                        <span className="rounded-full bg-purple-100 px-2 py-0.5 font-black text-purple-700">
   505→                          {d.tipo === "SELECCION_MULTIPLE"
   506→                            ? "🎯 Selección múltiple"
   507→                            : "✏️ Completar texto"}
   508→                        </span>
   509→                        <span className="font-bold text-purple-400">
   510→                          · {d.opciones.length} opciones
   511→                        </span>
   512→                      </div>
   513→                      <ul className="mt-2 space-y-0.5">
   514→                        {d.opciones.map((op) => (
   515→                          <li
   516→                            key={op.id}
   517→                            className="text-xs font-bold text-purple-500"
   518→                          >
   519→                            · {op.texto}
   520→                          </li>
   521→                        ))}
   522→                      </ul>
   523→                    </div>
   524→                    <button
   525→                      onClick={() => eliminarDesafio(d)}
   526→                      title="Eliminar desafío"
   527→                      className="rounded-xl bg-red-100 p-2 text-red-400 transition-colors hover:bg-red-500 hover:text-white"
   528→                    >
   529→                      🗑️
   530→                    </button>
   531→                  </div>
   532→                </div>
   533→              ))}
   534→            </div>
   535→          )}
   536→        </>
   537→      )}
   538→
   539→      {/* MODALES */}
   540→      {modalAsignatura && (
   541→        <CrearAsignaturaModal
   542→          onClose={() => setModalAsignatura(false)}
   543→          onCrear={crearAsignatura}
   544→        />
   545→      )}
   546→      {modalModulo && asignaturaSel && (
   547→        <CrearModuloModal
   548→          asignaturaNombre={asignaturaSel.nombre}

   746→function CrearDesafioModal({
   747→  onClose,
   748→  onCrear,
   749→}: {
   750→  onClose: () => void;
   751→  onCrear: (data: {
   752→    tipo: string;
   753→    pregunta: string;
   754→    puntos: number;
   755→    opciones: OpcionDraft[];
   756→  }) => Promise<void>;
   757→}) {
   758→  const [tipo, setTipo] = useState("SELECCION_MULTIPLE");
   759→  const [pregunta, setPregunta] = useState("");
   760→  const [puntos, setPuntos] = useState(10);
   761→  const [opciones, setOpciones] = useState<OpcionDraft[]>([
   762→    { texto: "", esCorrecta: false },
   763→    { texto: "", esCorrecta: false },
   764→  ]);
   765→  const [creando, setCreando] = useState(false);
   766→
   767→  const actualizarOpcion = (
   768→    i: number,
   769→    campo: keyof OpcionDraft,
   770→    valor: string | boolean,
   771→  ) => {
   772→    setOpciones((prev) =>
   773→      prev.map((o, idx) =>
   774→        idx === i
   775→          ? { ...o, [campo]: valor }
   776→          : campo === "esCorrecta" && valor
   777→            ? { ...o, esCorrecta: false }
   778→            : o,
   779→      ),
   780→    );
   781→  };
   782→
   783→  const agregarOpcion = () =>
   784→    setOpciones((prev) => [...prev, { texto: "", esCorrecta: false }]);
   785→
   786→  const quitarOpcion = (i: number) =>
   787→    setOpciones((prev) =>
   788→      prev.length <= 2 ? prev : prev.filter((_, idx) => idx !== i),
   789→    );
   790→
   791→  return (
   792→    <ModalShell title="Nuevo desafío" emoji="🎯" onClose={onClose}>
   793→      <form
   794→        onSubmit={async (e) => {
   795→          e.preventDefault();
   796→          if (!pregunta.trim()) return;
   797→          setCreando(true);
   798→          await onCrear({ tipo, pregunta, puntos, opciones });
   799→          setCreando(false);
   800→        }}
   801→        className="space-y-4"
   802→      >
   803→        <div>
   804→          <label className="mb-1.5 block text-sm font-black text-purple-700">
   805→            ❓ Pregunta
   806→          </label>
   807→          <textarea
   808→            value={pregunta}
   809→            onChange={(e) => setPregunta(e.target.value)}
   810→            placeholder="Ej: ¿Cuál es el resultado de 5 + 3?"
   811→            className="w-full rounded-2xl border-2 border-purple-100 bg-purple-50/50 px-4 py-3 font-bold text-purple-900 outline-none transition-all placeholder:text-purple-300 focus:border-violet-400 focus:bg-white focus:ring-4 focus:ring-violet-100"
   812→            rows={2}
   813→            autoFocus
   814→            maxLength={300}
   815→          />
   816→        </div>
   817→
   818→        <div className="grid grid-cols-2 gap-3">
   819→          <div>
   820→            <label className="mb-1.5 block text-sm font-black text-purple-700">
   821→              🎯 Tipo
   822→            </label>
   823→            <select
   824→              value={tipo}
   825→              onChange={(e) => setTipo(e.target.value)}
   826→              className="w-full rounded-2xl border-2 border-purple-100 bg-purple-50/50 px-4 py-3 font-bold text-purple-900 outline-none focus:border-violet-400 focus:bg-white focus:ring-4 focus:ring-violet-100"
   827→            >
   828→              <option value="SELECCION_MULTIPLE">🎯 Selección múltiple</option>
   829→              <option value="COMPLETAR_TEXTO">✏️ Completar texto</option>
   830→            </select>
   831→          </div>
   832→          <div>
   833→            <label className="mb-1.5 block text-sm font-black text-purple-700">
   834→              ⭐ Puntos
   835→            </label>
   836→            <input
   837→              type="number"
   838→              min={1}
   839→              max={100}
   840→              value={puntos}
   841→              onChange={(e) => setPuntos(Number(e.target.value) || 1)}
   842→              className="w-full rounded-2xl border-2 border-purple-100 bg-purple-50/50 px-4 py-3 text-center text-xl font-black text-purple-900 outline-none focus:border-violet-400 focus:bg-white focus:ring-4 focus:ring-violet-100"
   843→            />
   844→          </div>
   845→        </div>
   846→
   847→        <div>
   848→          <div className="mb-2 flex items-center justify-between">
   849→            <label className="text-sm font-black text-purple-700">
   850→              ✅ Opciones (marca la correcta)
   851→            </label>
   852→            <button
   853→              type="button"
   854→              onClick={agregarOpcion}
   855→              className="rounded-xl bg-violet-100 px-3 py-1 text-xs font-black text-violet-700 transition-colors hover:bg-violet-200"
   856→            >
   857→              + Añadir
   858→            </button>
   859→          </div>
   860→          <div className="space-y-2">
   861→            {opciones.map((op, i) => (
   862→              <div key={i} className="flex items-center gap-2">
   863→                <input
   864→                  type="radio"
   865→                  name="opcionCorrecta"
   866→                  checked={op.esCorrecta}
   867→                  onChange={() => actualizarOpcion(i, "esCorrecta", true)}
   868→                  className="h-5 w-5 accent-violet-600"
   869→                  title="Marcar como correcta"
   870→                />
   871→                <input
   872→                  type="text"
   873→                  value={op.texto}
   874→                  onChange={(e) => actualizarOpcion(i, "texto", e.target.value)}
   875→                  placeholder={`Opción ${i + 1}`}
   876→                  className="flex-1 rounded-xl border-2 border-purple-100 bg-purple-50/50 px-3 py-2 text-sm font-bold text-purple-900 outline-none focus:border-violet-400 focus:bg-white"
   877→                  maxLength={100}
   878→                />
   879→                {opciones.length > 2 && (
   880→                  <button
   881→                    type="button"
   882→                    onClick={() => quitarOpcion(i)}
   883→                    className="rounded-lg bg-red-100 p-1.5 text-red-400 hover:bg-red-500 hover:text-white"
   884→                  >
   885→                    ✕
   886→                  </button>
   887→                )}
   888→              </div>
   889→            ))}
   890→          </div>
   891→          <p className="mt-1.5 text-xs font-bold text-purple-400">
   892→            💡 Debe haber exactamente una opción marcada como correcta.
   893→          </p>
   894→        </div>
   895→
   896→        <button
   897→          type="submit"
   898→          disabled={creando}
   899→          className="btn-3d w-full rounded-2xl bg-gradient-to-r from-fuchsia-500 via-violet-500 to-purple-600 py-3.5 text-base font-black text-white shadow-xl shadow-violet-200 disabled:opacity-60"
   900→        >
   901→          {creando ? "⏳ Creando..." : "✨ ¡Crear desafío!"}
   902→        </button>
   903→      </form>
   904→    </ModalShell>
   905→  );
   906→}
