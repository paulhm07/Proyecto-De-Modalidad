1 button_3d_normal.xml — Estado normal

res/drawable/button_3d_normal.xml
<?xml version="1.0" encoding="utf-8"?>
<layer-list xmlns:android="http://schemas.android.com/apk/res/android">

    <!-- 1) Sombra exterior difuminada (base del relieve) -->
    <item>
        <shape android:shape="rectangle">
            <solid android:color="#1F000000" />
            <corners android:radius="28dp" />
        </shape>
    </item>

    <!-- 2) Capa intermedia de sombra desplazada hacia abajo -->
    <item android:top="2dp" android:left="2dp" android:right="2dp">
        <shape android:shape="rectangle">
            <solid android:color="#14000000" />
            <corners android:radius="26dp" />
        </shape>
    </item>

    <!-- 3) Cuerpo principal del botón con gradiente vibrante -->
    <item android:top="2dp" android:left="2dp" android:right="2dp" android:bottom="6dp">
        <shape android:shape="rectangle">
            <gradient
                android:startColor="#FFAB40"
                android:centerColor="#FF6E40"
                android:endColor="#F4511E"
                android:angle="270"
                android:type="linear" />
            <corners android:radius="24dp" />
            <stroke android:width="1.5dp" android:color="#FFE0B2" />
        </shape>
    </item>

    <!-- 4) Brillo superior interno (efecto plastilina brillante) -->
    <item android:top="4dp" android:left="6dp" android:right="6dp" android:bottom="12dp">
        <shape android:shape="rectangle">
            <gradient
                android:startColor="#80FFFFFF"
                android:centerColor="#33FFFFFF"
                android:endColor="#00FFFFFF"
                android:angle="90"
                android:type="linear" />
            <corners android:radius="20dp" />
        </shape>
    </item>

</layer-list>

3 button_3d_selector.xml — Selector de estados

res/drawable/button_3d_selector.xml
<?xml version="1.0" encoding="utf-8"?>
<selector xmlns:android="http://schemas.android.com/apk/res/android">

    <!-- Botón presionado (hundido) -->
    <item android:state_pressed="true"
          android:drawable="@drawable/button_3d_pressed" />

    <!-- Botón deshabilitado -->
    <item android:state_enabled="false"
          android:drawable="@drawable/button_3d_normal" />

    <!-- Estado por defecto (normal / levantado) -->
    <item android:drawable="@drawable/button_3d_normal" />

</selector>

4 activity_main_example.xml — Layout con tamaño táctil accesible

res/layout/activity_main.xml
<?xml version="1.0" encoding="utf-8"?>
<LinearLayout xmlns:android="http://schemas.android.com/apk/res/android"
    android:layout_width="match_parent"
    android:layout_height="match_parent"
    android:orientation="vertical"
    android:gravity="center"
    android:padding="24dp"
    android:background="#FFF8E1">

    <!--
        Botón infantil 3D estilo plastilina.
        - minHeight 72dp: objetivo táctil accesible para niños (>48dp mínimo)
        - stateListAnimator="@null": desactiva elevación Material por defecto
        - elevation="0dp": evita conflicto con sombra del sistema
    -->
    <Button
        android:id="@+id/btnJugar"
        android:layout_width="240dp"
        android:layout_height="72dp"
        android:minHeight="72dp"
        android:minWidth="160dp"
        android:background="@drawable/button_3d_selector"
        android:text="¡JUGAR!"
        android:textColor="#FFFFFF"
        android:textSize="24sp"
        android:textStyle="bold"
        android:fontFamily="sans-serif-black"
        android:letterSpacing="0.05"
        android:paddingStart="24dp"
        android:paddingEnd="24dp"
        android:stateListAnimator="@null"
        android:elevation="0dp"
        android:shadowColor="#BF360C"
        android:shadowDx="0"
        android:shadowDy="2"
        android:shadowRadius="3"
        android:textAllCaps="false"
        android:clickable="true"
        android:focusable="true" />

</LinearLayout>
✅ Características técnicas:
  • Compatibilidad: API 21+ (Android 5.0 Lollipop) sin dependencias externas
  • Esquinas redondeadas de 24dp (estilo Claymorphism)
  • Gradiente cálido de 3 colores (naranja → coral → naranja oscuro)
  • Sombra externa de 2 capas para relieve 3D realista
  • Brillo interno superior para efecto plastilina brillante
  • En estado presionado: sombra reducida + cuerpo desplazado 4dp hacia abajo
  • Tamaño táctil: 72dp de alto (recomendado para niños, mínimo 48dp)
